   279→export function SellerDealTracker() {
   280→  const { setSellerPanel, activeDeal, user } = useAppStore();
   281→  const mounted = useSyncExternalStore(emptySubscribe, () => true, () => false);
   282→
   283→  const [dealData, setDealData] = useState<DealData | null>(null);
   284→  const [loading, setLoading] = useState(true);
   285→  const [actionLoading, setActionLoading] = useState(false);
   286→
   287→  const fetchDeal = useCallback(async () => {
   288→    if (!activeDeal?.id) {
   289→      setLoading(false);
   290→      return;
   291→    }
   292→    setLoading(true);
   293→    try {
   294→      const res = await fetch(`/api/deals/${encodeURIComponent(activeDeal.id)}`);
   295→      if (res.ok) {
   296→        const data: DealData = await res.json();
   297→        setDealData(data);
   298→        if (data.status !== activeDeal.status) {
   299→          useAppStore.getState().setActiveDeal({
   300→            id: data.id,
   301→            title: data.title,
   302→            amount: data.amount,
   303→            status: data.status,
   304→            createdAt: data.createdAt,
   305→            buyerId: data.buyerId,
   306→            sellerId: data.sellerId || undefined,
   307→            creatorId: data.creatorId,
   308→            buyerName: data.buyer?.name,
   309→            sellerName: data.seller?.name,
   310→          });
   311→        }
   312→      }
   313→    } catch {
   314→      // Use store data as fallback
   315→    } finally {
   316→      setLoading(false);
   317→    }
   318→  }, [activeDeal?.id, activeDeal?.status]);
   319→
   320→  useEffect(() => {
   321→    fetchDeal();
   322→  }, [fetchDeal]);
   323→
   324→  if (!mounted) return null;
   325→
   326→  const status = dealData?.status || activeDeal?.status || 'created';
   327→  const activeStep = statusToActiveStep(status);
   328→  const isDisputed = status === 'disputed';
   329→  const isCancelled = status === 'cancelled' || status === 'rejected';
   330→  const isCompleted = status === 'completed';
   331→
   332→  const dealAmount = dealData?.amount ?? activeDeal?.amount ?? 0;
   333→  const dealTitle = dealData?.title || activeDeal?.title || 'ডিল';
   334→  const dealDate = (dealData?.createdAt || activeDeal?.createdAt)
   335→    ? new Date(dealData?.createdAt || activeDeal?.createdAt || '').toLocaleDateString('bn-BD', { year: 'numeric', month: 'short', day: 'numeric' })
   336→    : '---';
   337→  const buyerName = dealData?.buyer?.name || activeDeal?.buyerName || 'ক্রেতা';
   338→  const sellerName = dealData?.seller?.name || activeDeal?.sellerName || user?.name || 'বিক্রেতা';
   339→  const dealTerms = dealData?.terms;
   340→
   341→  const userId = user?.id;
   342→  const isSeller = userId === (dealData?.sellerId) || userId === activeDeal?.sellerId;
   343→
   344→  /* ── Action Handlers ── */
   345→  const handleDeliver = async () => {
   346→    if (!dealData) return;
   347→    setActionLoading(true);
   348→    try {
   349→      const res = await fetch('/api/deals/deliver', {
   350→        method: 'POST',
   351→        headers: { 'Content-Type': 'application/json' },
   352→        body: JSON.stringify({ dealId: dealData.id }),
   353→      });
   354→      if (res.ok) {
   355→        toast.success('কাজ সম্পন্ন হিসেবে চিহ্নিত হয়েছে! ক্রেতাকে জানানো হচ্ছে।');
   356→        fetchDeal();
   357→      } else {
   358→        const data = await res.json().catch(() => ({}));
   359→        toast.error(data.error || 'আপডেট ব্যর্থ হয়েছে');
   360→      }
   361→    } catch {
   362→      toast.error('নেটওয়ার্ক সমস্যা');
   363→    } finally {
   364→      setActionLoading(false);
   365→    }
   366→  };
   367→
   368→  const handleCancel = async () => {
   369→    if (!dealData) return;
   370→    setActionLoading(true);
   371→    try {
   372→      const res = await fetch('/api/deals/cancel', {
   373→        method: 'POST',
   374→        headers: { 'Content-Type': 'application/json' },
   375→        body: JSON.stringify({ dealId: dealData.id }),
   376→      });
   377→      if (res.ok) {
   378→        toast.success('ডিলটি বাতিল করা হয়েছে');
   379→        fetchDeal();
   380→      } else {
   381→        const data = await res.json().catch(() => ({}));
   382→        toast.error(data.error || 'বাতিল করতে সমস্যা');
   383→      }
   384→    } catch {
   385→      toast.error('নেটওয়ার্ক সমস্যা');
   386→    } finally {
   387→      setActionLoading(false);
   388→    }

   398→  }

   401→    <motion.div
   402→      initial={{ opacity: 0, y: 16 }}

   406→    >

   408→        {/* ── Header ── */}
   409→        <div className="flex items-center justify-between border-b px-4 py-3 sm:px-5" style={{ borderColor: 'var(--border)' }}>
   410→          <div className="flex items-center gap-3">
   411→            <button
   412→              onClick={() => setSellerPanel('active-deals')}
   413→              className="flex h-8 w-8 items-center justify-center rounded-lg transition-colors hover:bg-accent"
   414→              aria-label="ফিরে যান"
   415→            >
   416→              <ArrowLeft className="h-4 w-4" style={{ color: 'var(--foreground)' }} />
   417→            </button>
   418→            <div className="flex items-center gap-2">
   419→              <div className="flex h-8 w-8 items-center justify-center rounded-lg shadow-md" style={{ backgroundColor: '#84CC16' }}>
   420→                <span className="text-sm font-bold" style={{ color: '#18181b' }}>আ</span>

   423→                আমার ডিল
   424→              </span>
   425→            </div>
   426→          </div>
   427→
   428→          <div className="flex items-center gap-2">
   429→            <span className="hidden sm:block text-xs font-medium truncate max-w-[100px]" style={{ color: 'var(--muted-foreground)' }}>
   430→              {user?.name || 'বিক্রেতা'}
   431→            </span>
   432→            <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-xs font-bold" style={{ backgroundColor: '#84CC1625', color: '#84CC16' }}>

   438→        {/* ── Deal Info Content ── */}
