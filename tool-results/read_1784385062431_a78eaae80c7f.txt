    71→/* ─── Admin Chat Message Styles ─── */
    72→function adminChatBubble(msg: AdminChatMsg) {
    73→  const isAdmin = msg.role === 'admin';
    74→  const isSystem = msg.role === 'system';
    75→  const isBuyer = msg.role === 'buyer';
    76→  const isSeller = msg.role === 'seller';
    77→
    78→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    79→
    80→  const bubbleStyle = isSystem
    81→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    82→    : isAdmin
    83→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    84→    : isBuyer
    85→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    86→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    87→
    88→  const labelStyle = isSystem
    89→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    90→    : isAdmin
    91→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    92→    : isBuyer
    93→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    94→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    95→
    96→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    97→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    98→
    99→  return (
   100→    <div key={msg.id} className={`flex ${align}`}>
   101→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
   102→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
   103→        <p className="leading-relaxed">{msg.text}</p>
   104→      </div>
   105→    </div>
   106→  );

   442→function PaymentVerifyPanel() {
   443→  const [deals, setDeals] = useState<DealRow[]>([]);
   444→  const [loading, setLoading] = useState(true);
   445→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   446→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   447→  const [confirmDialog, setConfirmDialog] = useState<{
   448→    type: 'verify' | 'wrong_info' | 'cancel';
   449→    deal: DealRow;
   450→  } | null>(null);
   451→  const [editedAmount, setEditedAmount] = useState('');
   452→  const [isUpdatingAmount, setIsUpdatingAmount] = useState(false);
   453→
   454→  // Initialize edited amount when a deal is selected
   455→  useEffect(() => {
   456→    if (selectedDeal) {
   457→      setEditedAmount(String(selectedDeal.paymentAmount || selectedDeal.amount || ''));
   458→    }
   459→  }, [selectedDeal]);
   460→
   461→  const fetchDeals = useCallback(async () => {
   462→    try {
   463→      const res = await fetch('/api/admin/deals');
   464→      if (res.ok) {
   465→        const data = await res.json();
   466→        setDeals(data);
   467→      }
   468→    } catch {
   469→      // use fallback
   470→    } finally {
   471→      setLoading(false);
   472→    }
   473→  }, []);
   474→
   475→  useEffect(() => {
   476→    fetchDeals();
   477→  }, [fetchDeals]);
   478→
   479→  const handleVerify = async () => {
   480→    if (!confirmDialog) return;
   481→    const dealId = confirmDialog.deal.id;
   482→    setActionLoading(dealId);
   483→    try {
   484→      const res = await fetch('/api/admin/verify', {
   485→        method: 'POST',
   486→        headers: { 'Content-Type': 'application/json' },
   487→        body: JSON.stringify({ dealId }),
   488→      });
   489→      if (res.ok) {
   490→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   491→        setConfirmDialog(null);
   492→        setSelectedDeal(null);
   493→        fetchDeals();
   494→      } else {
   495→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   496→        setConfirmDialog(null);
   497→      }
   498→    } catch {
   499→      toast.error('সার্ভারে সমস্যা');
   500→      setConfirmDialog(null);
   501→    } finally {
   502→      setActionLoading(null);
   503→    }
   504→  };
   505→
   506→  const handleWrongInfo = async () => {
   507→    if (!confirmDialog) return;
   508→    const dealId = confirmDialog.deal.id;
   509→    setActionLoading(dealId);
   510→    try {
   511→      const res = await fetch('/api/admin/reject', {
   512→        method: 'POST',
   513→        headers: { 'Content-Type': 'application/json' },
   514→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   515→      });
   516→      if (res.ok) {
   517→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   518→        setConfirmDialog(null);
   519→        setSelectedDeal(null);
   520→        fetchDeals();
   521→      } else {
   522→        const data = await res.json().catch(() => ({}));
   523→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   524→        setConfirmDialog(null);
   525→      }
   526→    } catch {
   527→      toast.error('সার্ভারে সমস্যা');
   528→      setConfirmDialog(null);
   529→    } finally {
   530→      setActionLoading(null);
   531→    }
   532→  };
   533→
   534→  const handleCancelDeal = async () => {
   535→    if (!confirmDialog) return;
   536→    const dealId = confirmDialog.deal.id;
   537→    setActionLoading(dealId);
   538→    try {
   539→      const res = await fetch('/api/admin/reject', {
   540→        method: 'POST',
   541→        headers: { 'Content-Type': 'application/json' },
   542→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   543→      });
