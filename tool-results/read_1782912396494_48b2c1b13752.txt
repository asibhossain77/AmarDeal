    97→    97→function NotificationBell() {
    98→    98→  const user = useAppStore((s) => s.user);
    99→    99→  const setDashboardPanel = useAppStore((s) => s.setDashboardPanel);
   100→   100→  const setActiveDeal = useAppStore((s) => s.setActiveDeal);
   101→   101→  const [open, setOpen] = useState(false);
   102→   102→  const [notifications, setNotifications] = useState<NotifItem[]>([]);
   103→   103→  const [unreadCount, setUnreadCount] = useState(0);
   104→   104→  const [loading, setLoading] = useState(true);
   105→   105→  const dropdownRef = useRef<HTMLDivElement>(null);
   106→   106→
   107→   107→  const { onNotification, latestNotification } = useSocket(user?.id);
   108→   108→
   109→   109→  // Fetch notifications on mount
   110→   110→  const fetchNotifs = useCallback(async () => {
   111→   111→    if (!user?.id) return;
   112→   112→    try {
   113→   113→      const res = await fetch(`/api/notifications?userId=${user.id}`);
   114→   114→      if (res.ok) {
   115→   115→        const data = await res.json();
   116→   116→        setNotifications(data.notifications || []);
   117→   117→        setUnreadCount(data.unreadCount || 0);
   118→   118→      }
   119→   119→    } catch { /* silent */ }
   120→   120→    finally { setLoading(false); }
   121→   121→  }, [user?.id]);
   122→   122→
   123→   123→  useEffect(() => { fetchNotifs() }, [fetchNotifs]);
   124→   124→
   125→   125→  // Listen for real-time notifications
   126→   126→  useEffect(() => {
   127→   127→    const unsub = onNotification(() => {
   128→   128→      fetchNotifs();
   129→   129→      toast.success('নতুন নোটিফিকেশন এসেছে!', {
   130→   130→        description: 'ডিল সেকশনে দেখুন',
   131→   131→      });
   132→   132→    });
   133→   133→    return unsub;
   134→   134→  }, [onNotification, fetchNotifs]);
   135→   135→
   136→   136→  // Close dropdown on outside click
   137→   137→  useEffect(() => {
   138→   138→    const handleClick = (e: MouseEvent) => {
   139→   139→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   140→   140→        setOpen(false);
   141→   141→      }
   142→   142→    };
   143→   143→    document.addEventListener('mousedown', handleClick);
   144→   144→    return () => document.removeEventListener('mousedown', handleClick);
   145→   145→  }, []);
   146→   146→
   147→   147→  const handleMarkRead = async (notifId: string) => {
   148→   148→    try {
   149→   149→      await fetch(`/api/notifications/${notifId}/read`, { method: 'POST' });
   150→   150→      setNotifications((prev) =>
   151→   151→        prev.map((n) => (n.id === notifId ? { ...n, read: true } : n))
   152→   152→      );
   153→   153→      setUnreadCount((c) => Math.max(0, c - 1));
   154→   154→    } catch { /* silent */ }
   155→   155→  };
   156→   156→
   157→   157→  const handleMarkAllRead = async () => {
   158→   158→    if (!user?.id) return;
   159→   159→    try {
   160→   160→      await fetch('/api/notifications/all-read', {
   161→   161→        method: 'PUT',
   162→   162→        headers: { 'Content-Type': 'application/json' },
   163→   163→        body: JSON.stringify({ userId: user.id }),
   164→   164→      });
   165→   165→      setNotifications((prev) => prev.map((n) => ({ ...n, read: true })));
   166→   166→      setUnreadCount(0);
   167→   167→    } catch { /* silent */ }
   168→   168→  };
   169→   169→
   170→   170→  const handleNotifClick = (notif: NotifItem) => {
   171→   171→    handleMarkRead(notif.id);
   172→   172→    if (notif.dealId) {
   173→   173→      setActiveDeal({
   174→   174→        id: notif.dealId,
   175→   175→        title: notif.title,
   176→   176→        category: '',
   177→   177→        amount: 0,
   178→   178→        feePayer: '',
   179→   179→        status: 'created',
   180→   180→        createdAt: notif.createdAt,
   181→   181→      });
   182→   182→      setDashboardPanel('deal-detail');
   183→   183→    }
   184→   184→    setOpen(false);
   185→   185→  };
   186→   186→
   187→   187→  const timeAgo = (dateStr: string) => {
   188→   188→    const diff = Date.now() - new Date(dateStr).getTime();
   189→   189→    const mins = Math.floor(diff / 60000);
   190→   190→    if (mins < 1) return 'এইমাত্র';
   191→   191→    if (mins < 60) return `${mins} মিনিট আগে`;
   192→   192→    const hrs = Math.floor(mins / 60);
   193→   193→    if (hrs < 24) return `${hrs} ঘন্টা আগে`;
   194→   194→    const days = Math.floor(hrs / 24);
   195→   195→    return `${days} দিন আগে`;
   196→   196→  };
   197→   197→
   198→   198→  return (
   199→   199→    <div className="relative" ref={dropdownRef}>
   200→   200→      <button
   201→   201→        onClick={() => setOpen(!open)}
   202→   202→        className="relative flex h-10 w-10 items-center justify-center rounded-xl text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   203→   203→        aria-label="নোটিফিকেশন"
   204→   204→      >
   205→   205→        <Bell className="h-5 w-5" />
   206→   206→        {unreadCount > 0 && (
   207→   207→          <span className="absolute -top-0.5 -right-0.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-primary px-1 text-[10px] font-bold text-primary-foreground shadow-sm">
   208→   208→            {unreadCount > 99 ? '৯৯+' : unreadCount}
   209→   209→          </span>
   210→   210→        )}
   211→   211→      </button>
   212→   212→
   213→   213→      <AnimatePresence>
   214→   214→        {open && (
   215→   215→          <motion.div
   216→   216→            initial={{ opacity: 0, y: -8, scale: 0.95 }}
   217→   217→            animate={{ opacity: 1, y: 0, scale: 1 }}
   218→   218→            exit={{ opacity: 0, y: -8, scale: 0.95 }}
   219→   219→            transition={{ duration: 0.2 }}
   220→   220→            className="absolute right-0 top-12 z-50 w-80 sm:w-96 rounded-2xl bg-white shadow-2xl border border-border/50 dark:bg-zinc-900 overflow-hidden"
   221→   221→          >
   222→   222→            {/* Header */}
   223→   223→            <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   224→   224→              <h3 className="text-sm font-bold text-foreground">নোটিফিকেশন</h3>
   225→   225→              {unreadCount > 0 && (
   226→   226→                <button
   227→   227→                  onClick={handleMarkAllRead}
   228→   228→                  className="text-xs font-medium text-primary hover:underline"
   229→   229→                >
   230→   230→                  সব পড়া হয়েছে
   231→   231→                </button>
   232→   232→              )}
   233→   233→            </div>
   234→   234→
   235→   235→            {/* List */}
   236→   236→            <div className="max-h-80 overflow-y-auto">
   237→   237→              {loading ? (
   238→   238→                <div className="flex items-center justify-center py-10">
   239→   239→                  <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   240→   240→                </div>
   241→   241→              ) : notifications.length === 0 ? (
   242→   242→                <div className="flex flex-col items-center py-10 text-center">
   243→   243→                  <Bell className="h-8 w-8 text-muted-foreground/40 mb-2" />
   244→   244→                  <p className="text-sm text-muted-foreground">কোনো নোটিফিকেশন নেই</p>
   245→   245→                </div>
   246→   246→              ) : (
   247→   247→                notifications.map((notif) => (
   248→   248→                  <button
   249→   249→                    key={notif.id}
   250→   250→                    onClick={() => handleNotifClick(notif)}
   251→   251→                    className={`w-full text-left px-4 py-3 border-b border-border/30 transition-colors hover:bg-muted/30 last:border-0 ${
   252→   252→                      !notif.read ? 'bg-primary/5' : ''
   253→   253→                    }`}
   254→   254→                  >
   255→   255→                    <div className="flex items-start gap-3">
   256→   256→                      <div className={`mt-0.5 h-2 w-2 shrink-0 rounded-full ${
   257→   257→                        !notif.read ? 'bg-primary' : 'bg-transparent'
   258→   258→                      }`} />
   259→   259→                      <div className="min-w-0 flex-1">
   260→   260→                        <div className="flex items-center justify-between gap-2">
   261→   261→                          <p className={`text-sm font-semibold truncate ${!notif.read ? 'text-foreground' : 'text-muted-foreground'}`}>
   262→   262→                            {notif.title}
   263→   263→                          </p>
   264→   264→                          <span className="text-[10px] text-muted-foreground shrink-0">
   265→   265→                            {timeAgo(notif.createdAt)}
   266→   266→                          </span>
   267→   267→                        </div>
   268→   268→                        <p className="mt-0.5 text-xs text-muted-foreground line-clamp-2">
   269→   269→                          {notif.message}
   270→   270→                        </p>
   271→   271→                      </div>
   272→   272→                    </div>
   273→   273→                  </button>
   274→   274→                ))
   275→   275→              )}
   276→   276→            </div>
   277→   277→          </motion.div>
   278→   278→        )}
   279→   279→      </AnimatePresence>
   280→   280→    </div>
   281→   281→  );
   282→   282→}
