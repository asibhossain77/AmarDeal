   120→/**
   121→ * Maps a deal status to the active step index.
   122→ * Steps 0-4 correspond to the 5 STEPS.
   123→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
   124→ */
   125→function statusToActiveStep(status: string): number {
   126→  switch (status) {
   127→    case 'created': return 0;
   128→    case 'payment_pending': return 1;
   129→    case 'payment_verified': return 2;
   130→    case 'in_delivery': return 3;
   131→    case 'completed': return 4;
   132→    case 'cancelled':
   133→    case 'rejected': return -1;
   134→    case 'disputed': return 3;
   135→    default: return 0;
   136→  }
   137→}

   189→/** Convert ISO date string to Bengali time */
   190→function toBnTime(isoString: string): string {
   191→  try {
   192→    return new Date(isoString).toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
   193→  } catch {
   194→    return '';
   195→  }
   196→}
   197→
   198→/** Convert DB message to ChatMessage */
   199→function dbToChatMsg(m: { id: string; role: string | null; senderName: string | null; text: string; createdAt: string; senderId: string }): ChatMessage {
   200→  return {
   201→    id: m.id,
   202→    role: (m.role as MessageRole) || 'buyer',
   203→    senderName: m.senderName || 'অজানা',
   204→    text: m.text,
   205→    timestamp: toBnTime(m.createdAt),
   206→  };

   238→/**
   239→ * Horizontal 5-step progress bar (desktop).
   240→ * Uses parrot green #65A30D for active/completed steps.
   241→ */
   242→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   243→  if (isCancelled) {
   244→    return (
   245→      <div className="flex items-center justify-center py-6">
   246→        <div className="flex items-center gap-3 rounded-2xl bg-red-50 dark:bg-red-500/10 px-6 py-4 border border-red-200 dark:border-red-500/20">
   247→          <Ban className="h-8 w-8 text-red-500" />
   248→          <div>
   249→            <p className="text-base font-bold text-red-700 dark:text-red-400">ডিল বাতিল হয়েছে</p>
   250→            <p className="text-sm text-red-600/70 dark:text-red-400/70">এই ডিল আর সক্রিয় নয়</p>
   251→          </div>
   252→        </div>
   253→      </div>
   254→    );
   255→  }
   256→
   257→  return (
   258→    <div className="relative flex items-start justify-between py-2">
   259→      {/* Background track line */}
   260→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />
   261→      {/* Filled progress line */}
   262→      {activeStep >= 0 && (
   263→        <div
   264→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   265→          style={{
   266→            width: activeStep >= STEPS.length - 1
   267→              ? 'calc(100% - 40px)'
   268→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   269→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   270→          }}
   271→        />
   272→      )}
   273→
   274→      {STEPS.map((step, i) => {
   275→        const Icon = step.icon;
   276→        const isCompleted = i < activeStep;
   277→        const isActive = i === activeStep;
   278→
   279→        return (
   280→          <div
   281→            key={step.num}
   282→            className="relative z-10 flex flex-col items-center"
   283→            style={{ width: `${100 / STEPS.length}%` }}
   284→          >
   285→            <div className="relative">
   286→              {/* Pulse rings for active step */}
   287→              {isActive && (
   288→                <>
   289→                  <span
   290→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   291→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   292→                  />
   293→                  <span
   294→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   295→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   296→                  />
   297→                </>
   298→              )}
   299→              <div
   300→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   301→                style={{
   302→                  backgroundColor: isCompleted || isActive
   303→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)
   304→                    : 'var(--background)',
   305→                  borderColor: isCompleted || isActive
   306→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)
   307→                    : 'var(--border)',
   308→                  color: isCompleted ? '#fff' : (isActive && isDisputed ? '#EF4444' : (isActive ? '#fff' : 'var(--muted-foreground)')),
   309→                  boxShadow: (isCompleted || isActive) && !isDisputed ? PARROT_GREEN_GLOW : 'none',
   310→                }}
   311→              >
   312→                {isCompleted ? (
   313→                  <Check className="h-4 w-4" strokeWidth={3} />
   314→                ) : isActive && isDisputed ? (
   315→                  <AlertTriangle className="h-4 w-4" />
   316→                ) : (
   317→                  <Icon className="h-4 w-4" />
   318→                )}
   319→              </div>
   320→            </div>
   321→            <p
   322→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   323→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   324→            >
   325→              {step.label}
   326→            </p>
   327→          </div>
   328→        );
   329→      })}
   330→    </div>
   331→  );
   332→}
