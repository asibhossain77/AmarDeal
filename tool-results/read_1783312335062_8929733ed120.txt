   279→  const [loading, setLoading] = useState(true);

   281→
   282→  const fetchDeals = useCallback(async () => {
   283→    try {
   284→      const res = await fetch('/api/admin/deals');
   285→      if (res.ok) {
   286→        const data = await res.json();
   287→        setDeals(data);
   288→      }
   289→    } catch {
   290→      // use fallback
   291→    } finally {
   292→      setLoading(false);
   293→    }
   294→  }, []);
   295→
   296→  useEffect(() => {
   297→    fetchDeals();
   298→  }, [fetchDeals]);
   299→
   300→  const handleVerify = async (dealId: string) => {
   301→    setActionLoading(dealId);
   302→    try {
   303→      const res = await fetch('/api/admin/verify', {
   304→        method: 'POST',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify({ dealId }),
   307→      });

   309→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');

   312→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');

   315→      toast.error('সার্ভারে সমস্যা');

   317→      setActionLoading(null);

   338→      setActionLoading(null);
   339→    }

   342→  const pendingDeals = deals.filter(
   343→    (d) =>
   344→      d.status === 'created' ||
   345→      d.status === 'pending' ||

   348→

   359→            ভেরিফিকেশনের অপেক্ষমান ডিলসমূহ পর্যালোচনা করুন
   360→          </p>
   361→        </div>
   362→        <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   363→          {pendingDeals.length} পেন্ডিং
   364→        </Badge>
   365→      </div>
   366→
   367→      {/* Desktop Table */}
   368→      <SolidCard className="!p-0 overflow-hidden hidden lg:block">
   369→        {loading ? (
   370→          <div className="flex items-center justify-center py-16">
   371→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   372→          </div>
   373→        ) : pendingDeals.length === 0 ? (
   374→          <div className="flex flex-col items-center justify-center py-16 text-center">

   377→              সব পেমেন্ট ভেরিফাই হয়েছে
   378→            </p>
   379→            <p className="mt-1 text-xs text-muted-foreground">
   380→              বর্তমানে কোনো পেন্ডিং ডিল নেই
   381→            </p>
   382→          </div>
   383→        ) : (
   384→          <table className="w-full text-sm">
   385→            <thead>
