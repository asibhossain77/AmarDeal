    45→/* ─── Color Utility ─── */
    46→function getContrastColor(hex: string): string {
    47→  const r = parseInt(hex.slice(1, 3), 16);
    48→  const g = parseInt(hex.slice(3, 5), 16);
    49→  const b = parseInt(hex.slice(5, 7), 16);
    50→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
    51→  return luminance > 0.6 ? '#18181b' : '#ffffff';
    52→}
    53→
    54→/* ─── Method icon helper ─── */
    55→function getMethodIcon(name: string) {
    56→  const lower = name.toLowerCase();
    57→  if (lower.includes('bkash')) return Wallet;
    58→  if (lower.includes('nagad')) return CreditCard;
    59→  if (lower.includes('rocket')) return Wallet;
    60→  if (lower.includes('bank')) return Building2;
    61→  return CreditCard;
    62→}
