   178→/* ─── Admin Notification Bell ─── */
   179→function AdminNotificationBell() {
   180→  const user = useAppStore((s) => s.user);
   181→  const [open, setOpen] = useState(false);
   182→  const [notifications, setNotifications] = useState<any[]>([]);
   183→  const [unreadCount, setUnreadCount] = useState(0);
   184→  const [loading, setLoading] = useState(true);
   185→  const dropdownRef = useRef<HTMLDivElement>(null);
   186→
   187→  const { onNotification } = useSocket('admin');
   188→
   189→  const fetchNotifs = useCallback(async () => {
   190→    try {
   191→      const res = await fetch('/api/notifications?userId=admin');
   192→      if (res.ok) {
   193→        const data = await res.json();
   194→        setNotifications(data.notifications || []);
   195→        setUnreadCount(data.unreadCount || 0);
   196→      }
   197→    } catch { /* silent */ }
   198→    finally { setLoading(false); }
   199→  }, []);
   200→
   201→  useEffect(() => { fetchNotifs() }, [fetchNotifs]);
   202→
   203→  useEffect(() => {
   204→    const unsub = onNotification(() => {
   205→      fetchNotifs();
   206→      toast.success('নতুন নোটিফিকেশন!');
   207→    });
   208→    return unsub;
   209→  }, [onNotification, fetchNotifs]);
   210→
   211→  useEffect(() => {
   212→    const handleClick = (e: MouseEvent) => {
   213→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   214→        setOpen(false);
   215→      }
   216→    };
   217→    document.addEventListener('mousedown', handleClick);
   218→    return () => document.removeEventListener('mousedown', handleClick);
   219→  }, []);
   220→
   221→  return (
   222→    <div className="relative" ref={dropdownRef}>
   223→      <button
   224→        onClick={() => setOpen(!open)}
   225→        className="relative flex h-10 w-10 items-center justify-center rounded-xl text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   226→        aria-label="নোটিফিকেশন"
   227→      >
   228→        <Bell className="h-5 w-5" />
   229→        {unreadCount > 0 && (
   230→          <span className="absolute -top-0.5 -right-0.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-red-500 px-1 text-[10px] font-bold text-white shadow-sm">
   231→            {unreadCount > 99 ? '৯৯+' : unreadCount}
   232→          </span>
   233→        )}
   234→      </button>
   235→
   236→      <AnimatePresence>
   237→        {open && (
   238→          <motion.div
   239→            initial={{ opacity: 0, y: -8, scale: 0.95 }}
   240→            animate={{ opacity: 1, y: 0, scale: 1 }}
   241→            exit={{ opacity: 0, y: -8, scale: 0.95 }}
   242→            transition={{ duration: 0.2 }}
   243→            className="absolute right-0 top-12 z-50 w-80 rounded-2xl bg-white shadow-2xl border border-border/50 dark:bg-zinc-900 overflow-hidden"
   244→          >
   245→            <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   246→              <h3 className="text-sm font-bold text-foreground">অ্যাডমিন নোটিফিকেশন</h3>
   247→            </div>
   248→            <div className="max-h-72 overflow-y-auto">
   249→              {loading ? (
   250→                <div className="flex items-center justify-center py-10">
   251→                  <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   252→                </div>
   253→              ) : notifications.length === 0 ? (
   254→                <div className="flex flex-col items-center py-10 text-center">
   255→                  <Bell className="h-8 w-8 text-muted-foreground/40 mb-2" />
   256→                  <p className="text-sm text-muted-foreground">কোনো নোটিফিকেশন নেই</p>
   257→                </div>
   258→              ) : (
   259→                notifications.map((notif: any) => (
   260→                  <div
   261→                    key={notif.id}
   262→                    className={`px-4 py-3 border-b border-border/30 last:border-0 ${
   263→                      !notif.read ? 'bg-primary/5' : ''
   264→                    }`}
   265→                  >
   266→                    <div className="flex items-start gap-2">
   267→                      {!notif.read && <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />}
   268→                      <div className="min-w-0 flex-1">
   269→                        <p className="text-xs font-semibold text-foreground">{notif.title}</p>
   270→                        <p className="mt-0.5 text-[11px] text-muted-foreground line-clamp-2">{notif.message}</p>
   271→                      </div>
   272→                    </div>
   273→                  </div>
   274→                ))
   275→              )}
   276→            </div>
   277→          </motion.div>
   278→        )}
   279→      </AnimatePresence>
   280→    </div>
   281→  );
   282→}

   389→      if (res.ok) {

   391→        setDeals(data);
   392→      }

   394→      // use fallback
   395→    } finally {
   396→      setLoading(false);
   397→    }

   408→        method: 'POST',
