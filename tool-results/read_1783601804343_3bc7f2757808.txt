    65→/* ─── Admin Chat Message Styles ─── */
    66→function adminChatBubble(msg: AdminChatMsg) {
    67→  const isAdmin = msg.role === 'admin';
    68→  const isSystem = msg.role === 'system';
    69→  const isBuyer = msg.role === 'buyer';
    70→  const isSeller = msg.role === 'seller';
    71→
    72→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    73→
    74→  const bubbleStyle = isSystem
    75→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    76→    : isAdmin
    77→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    78→    : isBuyer
    79→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    80→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    81→
    82→  const labelStyle = isSystem
    83→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    84→    : isAdmin
    85→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    86→    : isBuyer
    87→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    88→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    89→
    90→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    91→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    92→
    93→  return (
    94→    <div key={msg.id} className={`flex ${align}`}>
    95→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
    96→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
    97→        <p className="leading-relaxed">{msg.text}</p>
    98→      </div>
    99→    </div>
   100→  );
   101→}

   397→    try {
   398→      const res = await fetch('/api/admin/deals');
   399→      if (res.ok) {
   400→        const data = await res.json();

   402→      }

   404→      // use fallback
   405→    } finally {
   406→      setLoading(false);
