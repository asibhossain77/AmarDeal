   118→/**
   119→ * Maps a deal status to the active step index.
   120→ * Steps 0-4 correspond to the 5 STEPS.
   121→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
   122→ */
   123→function statusToActiveStep(status: string): number {
   124→  switch (status) {
   125→    case 'created': return 0;
   126→    case 'payment_pending': return 1;
   127→    case 'payment_verified': return 2;
   128→    case 'in_delivery': return 3;
   129→    case 'completed': return 4;
   130→    case 'cancelled':
   131→    case 'rejected': return -1;
   132→    case 'disputed': return 3;
   133→    default: return 0;
   134→  }
   135→}

   188→/** Convert ISO date string to Bengali time */
   189→function toBnTime(isoString: string): string {
   190→  try {
   191→    return new Date(isoString).toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
   192→  } catch {
   193→    return '';
   194→  }
   195→}
   196→
   197→/** Convert DB message to ChatMessage */
   198→function dbToChatMsg(m: { id: string; role: string | null; senderName: string | null; text: string; createdAt: string; senderId: string }): ChatMessage {
   199→  return {
   200→    id: m.id,
   201→    role: (m.role as MessageRole) || 'system',
   202→    senderId: m.senderId,
   203→    senderName: m.senderName || 'অজানা',
   204→    text: m.text,
   205→    timestamp: toBnTime(m.createdAt),
   206→  };
   207→}

   240→/**
   241→ * Horizontal 5-step progress bar (desktop).
   242→ * Uses parrot green #65A30D for active/completed steps.
   243→ */
   244→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   245→  if (isCancelled) {
   246→    return (
   247→      <div className="flex items-center justify-center py-6">
   248→        <div className="flex items-center gap-3 rounded-2xl bg-red-50 dark:bg-red-500/10 px-6 py-4 border border-red-200 dark:border-red-500/20">
   249→          <Ban className="h-8 w-8 text-red-500" />
   250→          <div>
   251→            <p className="text-base font-bold text-red-700 dark:text-red-400">ডিল বাতিল হয়েছে</p>
   252→            <p className="text-sm text-red-600/70 dark:text-red-400/70">এই ডিল আর সক্রিয় নয়</p>
   253→          </div>
   254→        </div>
   255→      </div>
   256→    );
   257→  }
   258→
   259→  return (
   260→    <div className="relative flex items-start justify-between py-2">
   261→      {/* Background track line */}
   262→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />
   263→      {/* Filled progress line */}
   264→      {activeStep >= 0 && (
   265→        <div
   266→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   267→          style={{
   268→            width: activeStep >= STEPS.length - 1
   269→              ? 'calc(100% - 40px)'
   270→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   271→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   272→          }}
   273→        />
   274→      )}
   275→
   276→      {STEPS.map((step, i) => {
   277→        const Icon = step.icon;
   278→        const isCompleted = i < activeStep;
   279→        const isActive = i === activeStep;
   280→
   281→        return (
   282→          <div
   283→            key={step.num}
   284→            className="relative z-10 flex flex-col items-center"
   285→            style={{ width: `${100 / STEPS.length}%` }}
   286→          >
   287→            <div className="relative">
   288→              {/* Pulse rings for active step */}
   289→              {isActive && (
   290→                <>
   291→                  <span
   292→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   293→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   294→                  />
   295→                  <span
   296→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   297→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   298→                  />
   299→                </>
   300→              )}
   301→              <div
   302→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   303→                style={{
   304→                  backgroundColor: isCompleted || isActive
   305→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)
   306→                    : 'var(--background)',
   307→                  borderColor: isCompleted || isActive
   308→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)
   309→                    : 'var(--border)',
   310→                  color: isCompleted ? '#fff' : (isActive && isDisputed ? '#EF4444' : (isActive ? '#fff' : 'var(--muted-foreground)')),
   311→                  boxShadow: (isCompleted || isActive) && !isDisputed ? PARROT_GREEN_GLOW : 'none',
   312→                }}
   313→              >
   314→                {isCompleted ? (
   315→                  <Check className="h-4 w-4" strokeWidth={3} />
   316→                ) : isActive && isDisputed ? (
   317→                  <AlertTriangle className="h-4 w-4" />
   318→                ) : (
   319→                  <Icon className="h-4 w-4" />
   320→                )}
   321→              </div>
   322→            </div>
   323→            <p
   324→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   325→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   326→            >
   327→              {step.label}
   328→            </p>
   329→          </div>
   330→        );
   331→      })}
   332→    </div>
   333→  );
   334→}
