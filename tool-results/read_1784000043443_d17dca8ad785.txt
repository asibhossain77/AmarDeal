    47→/* ─── Color Utility ─── */
    48→function getContrastColor(hex: string): string {
    49→  const r = parseInt(hex.slice(1, 3), 16);
    50→  const g = parseInt(hex.slice(3, 5), 16);
    51→  const b = parseInt(hex.slice(5, 7), 16);
    52→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
    53→  return luminance > 0.6 ? '#18181b' : '#ffffff';
    54→}
    55→
    56→/* ─── Method icon/image helper ─── */
    57→function getMethodIcon(name: string) {
    58→  const lower = name.toLowerCase();
    59→  if (lower.includes('bkash')) return Wallet;
    60→  if (lower.includes('nagad')) return CreditCard;
    61→  if (lower.includes('rocket')) return Wallet;
    62→  if (lower.includes('bank')) return Building2;
    63→  return CreditCard;
    64→}
