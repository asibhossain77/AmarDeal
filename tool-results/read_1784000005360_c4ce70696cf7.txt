    68→/* ─── Admin Chat Message Styles ─── */
    69→function adminChatBubble(msg: AdminChatMsg) {
    70→  const isAdmin = msg.role === 'admin';
    71→  const isSystem = msg.role === 'system';
    72→  const isBuyer = msg.role === 'buyer';
    73→  const isSeller = msg.role === 'seller';
    74→
    75→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    76→
    77→  const bubbleStyle = isSystem
    78→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    79→    : isAdmin
    80→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    81→    : isBuyer
    82→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    83→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    84→
    85→  const labelStyle = isSystem
    86→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    87→    : isAdmin
    88→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    89→    : isBuyer
    90→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    91→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    92→
    93→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    94→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    95→
    96→  return (
    97→    <div key={msg.id} className={`flex ${align}`}>
    98→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
    99→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
   100→        <p className="leading-relaxed">{msg.text}</p>
   101→      </div>
   102→    </div>
   103→  );
   104→}

   412→function PaymentVerifyPanel() {
   413→  const [deals, setDeals] = useState<DealRow[]>([]);
   414→  const [loading, setLoading] = useState(true);
   415→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   416→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   417→  const [confirmDialog, setConfirmDialog] = useState<{
   418→    type: 'verify' | 'wrong_info' | 'cancel';
   419→    deal: DealRow;
   420→  } | null>(null);
   421→
   422→  const fetchDeals = useCallback(async () => {
   423→    try {
   424→      const res = await fetch('/api/admin/deals');
   425→      if (res.ok) {
   426→        const data = await res.json();
   427→        setDeals(data);
   428→      }
   429→    } catch {
   430→      // use fallback
   431→    } finally {
   432→      setLoading(false);
   433→    }
   434→  }, []);
   435→
   436→  useEffect(() => {
   437→    fetchDeals();
   438→  }, [fetchDeals]);
   439→
   440→  const handleVerify = async () => {
   441→    if (!confirmDialog) return;
   442→    const dealId = confirmDialog.deal.id;
   443→    setActionLoading(dealId);
   444→    try {
   445→      const res = await fetch('/api/admin/verify', {
   446→        method: 'POST',
   447→        headers: { 'Content-Type': 'application/json' },
   448→        body: JSON.stringify({ dealId }),
   449→      });
   450→      if (res.ok) {
   451→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   452→        setConfirmDialog(null);
   453→        setSelectedDeal(null);
   454→        fetchDeals();
   455→      } else {
   456→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   457→        setConfirmDialog(null);
   458→      }
   459→    } catch {
   460→      toast.error('সার্ভারে সমস্যা');
   461→      setConfirmDialog(null);
   462→    } finally {
   463→      setActionLoading(null);
   464→    }
   465→  };
   466→
   467→  const handleWrongInfo = async () => {
   468→    if (!confirmDialog) return;
   469→    const dealId = confirmDialog.deal.id;
   470→    setActionLoading(dealId);
   471→    try {
   472→      const res = await fetch('/api/admin/reject', {
   473→        method: 'POST',
   474→        headers: { 'Content-Type': 'application/json' },
   475→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   476→      });
   477→      if (res.ok) {
   478→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   479→        setConfirmDialog(null);
   480→        setSelectedDeal(null);
   481→        fetchDeals();
   482→      } else {
   483→        const data = await res.json().catch(() => ({}));
   484→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   485→        setConfirmDialog(null);
   486→      }
   487→    } catch {
   488→      toast.error('সার্ভারে সমস্যা');
   489→      setConfirmDialog(null);
   490→    } finally {
   491→      setActionLoading(null);
   492→    }
   493→  };
   494→
   495→  const handleCancelDeal = async () => {
   496→    if (!confirmDialog) return;
   497→    const dealId = confirmDialog.deal.id;
   498→    setActionLoading(dealId);
   499→    try {
   500→      const res = await fetch('/api/admin/reject', {
   501→        method: 'POST',
   502→        headers: { 'Content-Type': 'application/json' },
   503→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   504→      });
   505→      if (res.ok) {
   506→        toast.success('ডিল বাতিল করা হয়েছে');
   507→        setConfirmDialog(null);
   508→        setSelectedDeal(null);
   509→        fetchDeals();
   510→      } else {
   511→        const data = await res.json().catch(() => ({}));
   512→        toast.error(data.error || 'বাতিল করতে সমস্যা হয়েছে');
   513→        setConfirmDialog(null);
   514→      }
   515→    } catch {
   516→      toast.error('সার্ভারে সমস্যা');
   517→      setConfirmDialog(null);
   518→    } finally {
   519→      setActionLoading(null);
   520→    }
   521→  };
   522→
   523→  const pendingDeals = deals.filter(
   524→    (d) =>
   525→      d.status === 'created' ||
   526→      d.status === 'pending' ||
   527→      d.status === 'payment_pending'
   528→  );
   529→
   530→  /* ─── Detail View (shown when a deal is selected) ─── */
   531→  if (selectedDeal) {
   532→    const deal = selectedDeal;
   533→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   534→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   535→    const isActing = actionLoading === deal.id;
   536→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   537→      year: 'numeric',
   538→      month: 'short',
   539→      day: 'numeric',
   540→      hour: '2-digit',
   541→      minute: '2-digit',
   542→    });
   543→
   544→    return (
   545→      <div>
   546→        {/* Back button + header */}
   547→        <div className="mb-4 flex items-center gap-2">
   548→          <button
   549→            onClick={() => setSelectedDeal(null)}
   550→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
