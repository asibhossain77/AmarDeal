    55→/* ─── Known Gateway Defaults ─── */
    56→function getGatewayColor(name: string): string {
    57→  const n = name.toLowerCase();
    58→  if (n.includes('bkash')) return '#E2136E';
    59→  if (n.includes('nagad')) return '#F6921E';
    60→  if (n.includes('rocket')) return '#8C3494';
    61→  if (n.includes('bank')) return '#0033A0';
    62→  if (n.includes('upay')) return '#1A9F4B';
    63→  if (n.includes('dutch') || n.includes('bangla')) return '#FF6600';
    64→  return '#84CC16';
    65→}
    66→
    67→function getGatewayIcon(name: string) {
    68→  const n = name.toLowerCase();
    69→  if (n.includes('bank')) return '🏦';
    70→  return '💳';
    71→}

    91→function ColorPickerRow({
    92→  color,
    93→  onChange,
    94→  label,
    95→}: {
    96→  color: string;
    97→  onChange: (c: string) => void;
    98→  label?: string;
    99→}) {
   100→  return (
   101→    <div className="space-y-1.5">
   102→      {label && <Label className="text-sm font-medium">{label}</Label>}
   103→      <div className="flex items-center gap-3">
   104→        <div className="relative">
   105→          <input
   106→            type="color"
   107→            value={color}
   108→            onChange={(e) => onChange(e.target.value)}
   109→            className="h-10 w-14 cursor-pointer rounded-lg border border-border p-1 bg-transparent"
   110→          />
   111→        </div>
   112→        <Input
   113→          value={color}
   114→          onChange={(e) => {
   115→            const v = e.target.value;
   116→            if (/^#[0-9a-fA-F]{0,6}$/.test(v)) {
   117→              onChange(v.length === 7 ? v : color);

   120→          className="w-32 h-10 font-mono text-sm"
   121→        />
   122→        <div
   123→          className="h-10 w-20 rounded-xl border border-border/50 shrink-0"
   124→          style={{ backgroundColor: color }}
   125→        />
   126→      </div>
   127→      {/* Quick color presets */}
   128→      <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   129→        {['#E2136E', '#F6921E', '#8C3494', '#0033A0', '#1A9F4B', '#84CC16', '#65A30D', '#2563EB', '#DC2626', '#7C3AED'].map((c) => (

   131→            key={c}

   135→            style={{
   136→              backgroundColor: c,
   137→              borderColor: color === c ? 'var(--foreground)' : 'transparent',
   138→            }}

   145→}

   729→function getContrastColor(hex: string): string {
   730→  const r = parseInt(hex.slice(1, 3), 16);
   731→  const g = parseInt(hex.slice(3, 5), 16);
   732→  const b = parseInt(hex.slice(5, 7), 16);
   733→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   734→  return luminance > 0.6 ? '#18181b' : '#ffffff';
   735→}
