   656→function getContrastColor(hex: string): string {
   657→  const r = parseInt(hex.slice(1, 3), 16);
   658→  const g = parseInt(hex.slice(3, 5), 16);
   659→  const b = parseInt(hex.slice(5, 7), 16);
   660→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   661→  return luminance > 0.6 ? '#18181b' : '#ffffff';
   662→}
