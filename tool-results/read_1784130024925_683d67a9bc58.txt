   574→   574→export async function sendEmail(to: string, payload: EmailPayload): Promise<void> {
   575→   575→  const settings = await loadEmailSettings(); // refresh cache if stale
   576→   576→  const transporter = getTransporter(settings);
   577→   577→  if (!transporter) throw new Error('BREVO_SMTP_KEY সেট করা নেই। .env ফাইলে বা অ্যাডমিন প্যানেলে যোগ করুন।');
   578→   578→
   579→   579→  await transporter.sendMail({
   580→   580→    from: `${s('email_from_name')} <${getFromAddress(settings)}>`,
   581→   581→    to,
   582→   582→    subject: payload.subject,
   583→   583→    html: payload.html,
   584→   584→  });
   585→   585→  console.log(`[EMAIL SENT] → ${to}: ${payload.subject}`);
   586→   586→}
