    19→type EmailSettings = Record<string, string>;
    20→
    21→/* In-memory cache */
    22→let _settingsCache: EmailSettings | null = null;
    23→let _settingsLoadedAt = 0;
    24→const SETTINGS_TTL = 5 * 60 * 1000; // 5 min
    25→
    26→/** Load email settings from DB (call before sending) */
    27→export async function loadEmailSettings(): Promise<EmailSettings> {
    28→  const now = Date.now();
    29→  if (_settingsCache && now - _settingsLoadedAt < SETTINGS_TTL) return _settingsCache;
    30→  try {
    31→    const { db } = await import('@/lib/db');
    32→    const rows = await db.platformSetting.findMany({
    33→      where: { key: { in: EMAIL_SETTING_KEYS } },
    34→    });
    35→    const overrides: EmailSettings = {};
    36→    for (const r of rows) overrides[r.key] = r.value;
    37→    _settingsCache = { ...DEFAULTS, ...overrides };
    38→    _settingsLoadedAt = now;
    39→  } catch (e) {
    40→    console.error('[EMAIL SETTINGS] DB load failed, using defaults', e);
    41→    _settingsCache = { ...DEFAULTS };
    42→    _settingsLoadedAt = now;
    43→  }
    44→  return _settingsCache;
    45→}

    49→  return _settingsCache?.[key] ?? DEFAULTS[key];

    64→  const smtpKey = settings?.brevo_smtp_key || process.env.BREVO_SMTP_KEY;
    65→  if (!smtpKey) return null;

    67→    const smtpUser = settings?.brevo_smtp_user || process.env.BREVO_SMTP_USER || '';

    70→      port: 587,
    71→      secure: false,

    73→    });
    74→  }

   572→type EmailPayload = { subject: string; html: string };
   573→
   574→export async function sendEmail(to: string, payload: EmailPayload): Promise<void> {
   575→  const settings = await loadEmailSettings(); // refresh cache if stale
   576→  const transporter = getTransporter(settings);
   577→  if (!transporter) throw new Error('BREVO_SMTP_KEY সেট করা নেই। .env ফাইলে বা অ্যাডমিন প্যানেলে যোগ করুন।');
   578→
   579→  await transporter.sendMail({
   580→    from: `${s('email_from_name')} <${getFromAddress(settings)}>`,
   581→    to,
   582→    subject: payload.subject,
   583→    html: payload.html,
   584→  });
   585→  console.log(`[EMAIL SENT] → ${to}: ${payload.subject}`);
   586→}
   587→
   588→export function fireEmails(emails: Array<{ to: string; payload: EmailPayload }>) {
   589→  for (const e of emails) {
   590→    sendEmail(e.to, e.payload).catch((err) => {
   591→      console.error(`[EMAIL ERROR] → ${e.to}:`, err);
   592→    });
   593→  }
