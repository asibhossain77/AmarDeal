    41→    41→/* ─── Color Utility ─── */
    42→    42→function getContrastColor(hex: string): string {
    43→    43→  const r = parseInt(hex.slice(1, 3), 16);
    44→    44→  const g = parseInt(hex.slice(3, 5), 16);
    45→    45→  const b = parseInt(hex.slice(5, 7), 16);
    46→    46→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
    47→    47→  return luminance > 0.6 ? '#18181b' : '#ffffff';
    48→    48→}

    50→    50→/* ─── Method icon helper ─── */
    51→    51→function getMethodIcon(name: string) {
    52→    52→  const lower = name.toLowerCase();
    53→    53→  if (lower.includes('bkash')) return Wallet;
    54→    54→  if (lower.includes('nagad')) return CreditCard;
    55→    55→  if (lower.includes('rocket')) return Wallet;
    56→    56→  if (lower.includes('bank')) return Building2;
    57→    57→  return CreditCard;
    58→    58→}
