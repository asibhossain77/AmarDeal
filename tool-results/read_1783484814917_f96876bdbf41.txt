    46→/* ─── Color Utility ─── */
    47→function getContrastColor(hex: string): string {
    48→  const r = parseInt(hex.slice(1, 3), 16);
    49→  const g = parseInt(hex.slice(3, 5), 16);
    50→  const b = parseInt(hex.slice(5, 7), 16);
    51→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
    52→  return luminance > 0.6 ? '#18181b' : '#ffffff';
    53→}
    54→
    55→/* ─── Method icon/image helper ─── */
    56→function getMethodIcon(name: string) {
    57→  const lower = name.toLowerCase();
    58→  if (lower.includes('bkash')) return Wallet;
    59→  if (lower.includes('nagad')) return CreditCard;
    60→  if (lower.includes('rocket')) return Wallet;
    61→  if (lower.includes('bank')) return Building2;
    62→  return CreditCard;
    63→}
