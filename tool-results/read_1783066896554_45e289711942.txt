    63→/**

    65→ * Steps 0-4 correspond to the 5 STEPS.
    66→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).

    68→function statusToActiveStep(status: string): number {
    69→  switch (status) {
    70→    case 'created': return 0;
    71→    case 'Pending_Verification': return 1;
    72→    case 'Pending_Payment': return 1;
    73→    case 'Payment_Verified': return 2;
    74→    case 'in_delivery': return 3;
    75→    case 'completed': return 4;
    76→    case 'cancelled':
    77→    case 'rejected': return -1;
    78→    case 'disputed': return 3;
    79→    default: return 0;
    80→  }
    81→}

   193→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   194→  if (isCancelled) {

   209→    <div className="relative flex items-start justify-between py-2">
   210→      {/* Background track line */}

   214→        <div
   215→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   216→          style={{
   217→            width: activeStep >= STEPS.length - 1
   218→              ? 'calc(100% - 40px)'
   219→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   220→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   221→          }}
   222→        />
   223→      )}
   224→
   225→      {STEPS.map((step, i) => {
   226→        const Icon = step.icon;
   227→        const isCompleted = i < activeStep;
   228→        const isActive = i === activeStep;
   229→
   230→        return (
   231→          <div
   232→            key={step.num}
   233→            className="relative z-10 flex flex-col items-center"
   234→            style={{ width: `${100 / STEPS.length}%` }}

   236→            <div className="relative">

   239→                <>
   240→                  <span
   241→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   242→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   243→                  />
   244→                  <span
   245→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"

   251→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"

   253→                  backgroundColor: isCompleted || isActive
   254→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)

   257→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)

   264→                  <Check className="h-4 w-4" strokeWidth={3} />

   268→                  <Icon className="h-4 w-4" />
   269→                )}
   270→              </div>
   271→            </div>
   272→            <p
   273→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   274→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   275→            >
   276→              {step.label}
   277→            </p>
   278→          </div>
   279→        );
   280→      })}
   281→    </div>
   282→  );
   283→}
