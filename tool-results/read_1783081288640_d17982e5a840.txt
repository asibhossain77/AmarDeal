   349→   349→  }, [activeDeal?.id]);

   362→   362→

   365→   365→  const isDisputed = status === 'disputed';
   366→   366→  const isCancelled = status === 'cancelled';
   367→   367→  const isHardRejected = status === 'rejected';
   368→   368→  const isPendingPayment = status === 'Pending_Payment';
   369→   369→  const isCompleted = status === 'completed';
   370→   370→  const canRepay = isPendingPayment || isHardRejected;
   371→   371→
   372→   372→  const dealAmount = dealData?.amount ?? activeDeal?.amount ?? 0;
   373→   373→  const dealTitle = dealData?.title || activeDeal?.title || 'ডিল';
   374→   374→  const dealDate = (dealData?.createdAt || activeDeal?.createdAt)
   375→   375→    ? new Date(dealData?.createdAt || activeDeal?.createdAt || '').toLocaleDateString('bn-BD', { year: 'numeric', month: 'short', day: 'numeric' })
   376→   376→    : '---';
   377→   377→  const buyerName = dealData?.buyer?.name || activeDeal?.buyerName || user?.name || 'ক্রেতা';
   378→   378→  const sellerName = dealData?.seller?.name || activeDeal?.sellerName || 'বিক্রেতা';
   379→   379→  const dealTerms = dealData?.terms;
   380→   380→  const rejectionReason = dealData?.rejectionReason || activeDeal?.rejectionReason || null;
   381→   381→
   382→   382→  /* ── Action Handlers ── */
   383→   383→  const handlePayNow = () => {
   384→   384→    setDashboardPanel('payment');

   386→   386→

   390→   390→    try {
