    71→function adminChatBubble(msg: AdminChatMsg) {
    72→  const isAdmin = msg.role === 'admin';
    73→  const isSystem = msg.role === 'system';
    74→  const isBuyer = msg.role === 'buyer';
    75→  const isSeller = msg.role === 'seller';
    76→
    77→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    78→
    79→  const bubbleStyle = isSystem
    80→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    81→    : isAdmin
    82→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    83→    : isBuyer
    84→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    85→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    86→
    87→  const labelStyle = isSystem
    88→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    89→    : isAdmin
    90→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    91→    : isBuyer
    92→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    93→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    94→
    95→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    96→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    97→
    98→  return (
    99→    <div key={msg.id} className={`flex ${align}`}>
   100→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>

   441→function PaymentVerifyPanel() {
   442→  const [deals, setDeals] = useState<DealRow[]>([]);
   443→  const [loading, setLoading] = useState(true);
   444→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   445→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   446→  const [confirmDialog, setConfirmDialog] = useState<{
   447→    type: 'verify' | 'wrong_info' | 'cancel';
   448→    deal: DealRow;
   449→  } | null>(null);
   450→  const [editedAmount, setEditedAmount] = useState('');
   451→  const [isUpdatingAmount, setIsUpdatingAmount] = useState(false);
   452→
   453→  // Initialize edited amount when a deal is selected
   454→  useEffect(() => {
   455→    if (selectedDeal) {
   456→      setEditedAmount(String(selectedDeal.paymentAmount || selectedDeal.amount || ''));
   457→    }
   458→  }, [selectedDeal]);
   459→
   460→  const fetchDeals = useCallback(async () => {
   461→    try {
   462→      const res = await fetch('/api/admin/deals');
   463→      if (res.ok) {
   464→        const data = await res.json();
   465→        setDeals(data);
   466→      }
   467→    } catch {
   468→      // use fallback
   469→    } finally {
   470→      setLoading(false);
   471→    }
   472→  }, []);
   473→
   474→  useEffect(() => {
   475→    fetchDeals();
   476→  }, [fetchDeals]);
   477→
   478→  const handleVerify = async () => {
   479→    if (!confirmDialog) return;
   480→    const dealId = confirmDialog.deal.id;
   481→    setActionLoading(dealId);
   482→    try {
   483→      const res = await fetch('/api/admin/verify', {
   484→        method: 'POST',
   485→        headers: { 'Content-Type': 'application/json' },
   486→        body: JSON.stringify({ dealId }),
   487→      });
   488→      if (res.ok) {
   489→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   490→        setConfirmDialog(null);
   491→        setSelectedDeal(null);
   492→        fetchDeals();
   493→      } else {
   494→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   495→        setConfirmDialog(null);
   496→      }
   497→    } catch {
   498→      toast.error('সার্ভারে সমস্যা');
   499→      setConfirmDialog(null);
   500→    } finally {
   501→      setActionLoading(null);
   502→    }
   503→  };
   504→
   505→  const handleWrongInfo = async () => {
   506→    if (!confirmDialog) return;
   507→    const dealId = confirmDialog.deal.id;
   508→    setActionLoading(dealId);
   509→    try {
   510→      const res = await fetch('/api/admin/reject', {
   511→        method: 'POST',
   512→        headers: { 'Content-Type': 'application/json' },
   513→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   514→      });
   515→      if (res.ok) {
   516→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   517→        setConfirmDialog(null);
   518→        setSelectedDeal(null);
   519→        fetchDeals();
   520→      } else {
   521→        const data = await res.json().catch(() => ({}));
   522→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   523→        setConfirmDialog(null);
   524→      }
   525→    } catch {
   526→      toast.error('সার্ভারে সমস্যা');
   527→      setConfirmDialog(null);
   528→    } finally {
   529→      setActionLoading(null);
   530→    }
   531→  };
   532→
   533→  const handleCancelDeal = async () => {
   534→    if (!confirmDialog) return;
   535→    const dealId = confirmDialog.deal.id;
   536→    setActionLoading(dealId);
   537→    try {
   538→      const res = await fetch('/api/admin/reject', {
   539→        method: 'POST',
   540→        headers: { 'Content-Type': 'application/json' },
   541→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   542→      });
   543→      if (res.ok) {
