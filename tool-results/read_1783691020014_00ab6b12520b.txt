    65→/* ─── Admin Chat Message Styles ─── */
    66→function adminChatBubble(msg: AdminChatMsg) {
    67→  const isAdmin = msg.role === 'admin';
    68→  const isSystem = msg.role === 'system';
    69→  const isBuyer = msg.role === 'buyer';
    70→  const isSeller = msg.role === 'seller';
    71→
    72→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    73→
    74→  const bubbleStyle = isSystem
    75→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    76→    : isAdmin
    77→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    78→    : isBuyer
    79→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    80→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    81→
    82→  const labelStyle = isSystem
    83→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    84→    : isAdmin
    85→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    86→    : isBuyer
    87→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    88→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    89→
    90→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    91→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    92→
    93→  return (
    94→    <div key={msg.id} className={`flex ${align}`}>
    95→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
    96→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
    97→        <p className="leading-relaxed">{msg.text}</p>
    98→      </div>
    99→    </div>
   100→  );
   101→}

   391→function PaymentVerifyPanel() {
   392→  const [deals, setDeals] = useState<DealRow[]>([]);
   393→  const [loading, setLoading] = useState(true);
   394→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   395→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   396→  const [confirmDialog, setConfirmDialog] = useState<{
   397→    type: 'verify' | 'wrong_info' | 'cancel';
   398→    deal: DealRow;
   399→  } | null>(null);
   400→
   401→  const fetchDeals = useCallback(async () => {
   402→    try {
   403→      const res = await fetch('/api/admin/deals');
   404→      if (res.ok) {
   405→        const data = await res.json();
   406→        setDeals(data);
   407→      }
   408→    } catch {
   409→      // use fallback
   410→    } finally {
   411→      setLoading(false);
   412→    }
   413→  }, []);
   414→
   415→  useEffect(() => {
   416→    fetchDeals();
   417→  }, [fetchDeals]);
   418→
   419→  const handleVerify = async () => {
   420→    if (!confirmDialog) return;
   421→    const dealId = confirmDialog.deal.id;
   422→    setActionLoading(dealId);
   423→    try {
   424→      const res = await fetch('/api/admin/verify', {
   425→        method: 'POST',
   426→        headers: { 'Content-Type': 'application/json' },
   427→        body: JSON.stringify({ dealId }),
   428→      });
   429→      if (res.ok) {
   430→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   431→        setConfirmDialog(null);
   432→        setSelectedDeal(null);
   433→        fetchDeals();
   434→      } else {
   435→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   436→        setConfirmDialog(null);
   437→      }
   438→    } catch {
   439→      toast.error('সার্ভারে সমস্যা');
   440→      setConfirmDialog(null);
   441→    } finally {
   442→      setActionLoading(null);
   443→    }
   444→  };
   445→
   446→  const handleWrongInfo = async () => {
   447→    if (!confirmDialog) return;
   448→    const dealId = confirmDialog.deal.id;
   449→    setActionLoading(dealId);
   450→    try {
   451→      const res = await fetch('/api/admin/reject', {
   452→        method: 'POST',
   453→        headers: { 'Content-Type': 'application/json' },
   454→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   455→      });
   456→      if (res.ok) {
   457→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   458→        setConfirmDialog(null);
   459→        setSelectedDeal(null);
   460→        fetchDeals();
   461→      } else {
   462→        const data = await res.json().catch(() => ({}));
   463→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   464→        setConfirmDialog(null);
   465→      }
   466→    } catch {
   467→      toast.error('সার্ভারে সমস্যা');
   468→      setConfirmDialog(null);
   469→    } finally {
   470→      setActionLoading(null);
   471→    }
   472→  };
   473→
   474→  const handleCancelDeal = async () => {
   475→    if (!confirmDialog) return;
   476→    const dealId = confirmDialog.deal.id;
   477→    setActionLoading(dealId);
   478→    try {
   479→      const res = await fetch('/api/admin/reject', {
   480→        method: 'POST',
   481→        headers: { 'Content-Type': 'application/json' },
   482→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   483→      });
   484→      if (res.ok) {
   485→        toast.success('ডিল বাতিল করা হয়েছে');
   486→        setConfirmDialog(null);
   487→        setSelectedDeal(null);
   488→        fetchDeals();
   489→      } else {
   490→        const data = await res.json().catch(() => ({}));
   491→        toast.error(data.error || 'বাতিল করতে সমস্যা হয়েছে');
   492→        setConfirmDialog(null);
   493→      }
   494→    } catch {
   495→      toast.error('সার্ভারে সমস্যা');
   496→      setConfirmDialog(null);
   497→    } finally {
   498→      setActionLoading(null);
   499→    }
   500→  };
   501→
   502→  const pendingDeals = deals.filter(
   503→    (d) =>
   504→      d.status === 'created' ||
   505→      d.status === 'pending' ||
   506→      d.status === 'payment_pending'
   507→  );
   508→
   509→  /* ─── Detail View (shown when a deal is selected) ─── */
   510→  if (selectedDeal) {
   511→    const deal = selectedDeal;
   512→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   513→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   514→    const isActing = actionLoading === deal.id;
   515→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   516→      year: 'numeric',
   517→      month: 'short',
   518→      day: 'numeric',
   519→      hour: '2-digit',
   520→      minute: '2-digit',
   521→    });
   522→
   523→    return (
   524→      <div>
   525→        {/* Back button + header */}
   526→        <div className="mb-4 flex items-center gap-2">
   527→          <button
   528→            onClick={() => setSelectedDeal(null)}
   529→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   530→          >
   531→            <ArrowLeft className="h-4 w-4" />
   532→            ফিরে যান
   533→          </button>
   534→        </div>
   535→
   536→        <SolidCard className="!p-0 overflow-hidden">
   537→          {/* Header */}
   538→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   539→            <div className="flex items-center justify-between mb-2">
   540→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   541→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   542→                পেন্ডিং
   543→              </Badge>
   544→            </div>
   545→            <p className="text-base font-bold text-foreground">{userName}</p>
   546→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
