   289→
   290→function PaymentVerifyPanel() {
   291→  const [deals, setDeals] = useState<DealRow[]>([]);
   292→  const [loading, setLoading] = useState(true);
   293→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   294→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   295→
   296→  const fetchDeals = useCallback(async () => {
   297→    try {
   298→      const res = await fetch('/api/admin/deals');
   299→      if (res.ok) {
   300→        const data = await res.json();
   301→        setDeals(data);
   302→      }
   303→    } catch {
   304→      // use fallback
   305→    } finally {
   306→      setLoading(false);
   307→    }
   308→  }, []);
   309→
   310→  useEffect(() => {
   311→    fetchDeals();
   312→  }, [fetchDeals]);
   313→
   314→  const handleVerify = async (dealId: string) => {
   315→    setActionLoading(dealId);
   316→    try {
   317→      const res = await fetch('/api/admin/verify', {
   318→        method: 'POST',
   319→        headers: { 'Content-Type': 'application/json' },
   320→        body: JSON.stringify({ dealId }),
   321→      });
   322→      if (res.ok) {
   323→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   324→        setSelectedDeal(null);
   325→        fetchDeals();
   326→      } else {
   327→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   328→      }
   329→    } catch {
   330→      toast.error('সার্ভারে সমস্যা');
   331→    } finally {
   332→      setActionLoading(null);
   333→    }
   334→  };
   335→
   336→  const handleReject = async (dealId: string) => {
   337→    setActionLoading(dealId);
   338→    try {
   339→      const res = await fetch('/api/admin/reject', {
   340→        method: 'POST',
   341→        headers: { 'Content-Type': 'application/json' },
   342→        body: JSON.stringify({ dealId }),
   343→      });
   344→      if (res.ok) {
   345→        toast.success('ডিল রিজেক্ট করা হয়েছে');
   346→        setSelectedDeal(null);
   347→        fetchDeals();
   348→      } else {
   349→        toast.error('রিজেক্ট ব্যর্থ হয়েছে');
   350→      }
   351→    } catch {
   352→      toast.error('সার্ভারে সমস্যা');
   353→    } finally {
   354→      setActionLoading(null);
   355→    }
   356→  };
   357→
   358→  const pendingDeals = deals.filter(
   359→    (d) =>
   360→      d.status === 'created' ||
   361→      d.status === 'pending' ||
   362→      d.status === 'payment_pending'
   363→  );
   364→
   365→  /* ─── Detail View (shown when a deal is selected) ─── */
   366→  if (selectedDeal) {
   367→    const deal = selectedDeal;
   368→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   369→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   370→    const isActing = actionLoading === deal.id;
   371→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   372→      year: 'numeric',
   373→      month: 'short',
   374→      day: 'numeric',
   375→      hour: '2-digit',
   376→      minute: '2-digit',
   377→    });
   378→
   379→    return (
   380→      <div>
   381→        {/* Back button + header */}
   382→        <div className="mb-4 flex items-center gap-2">
   383→          <button
   384→            onClick={() => setSelectedDeal(null)}
   385→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   386→          >
   387→            <ArrowLeft className="h-4 w-4" />
   388→            ফিরে যান
   389→          </button>
   390→        </div>
   391→
   392→        <SolidCard className="!p-0 overflow-hidden">
   393→          {/* Header */}
   394→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   395→            <div className="flex items-center justify-between mb-2">
   396→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   397→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   398→                পেন্ডিং
   399→              </Badge>
   400→            </div>
   401→            <p className="text-base font-bold text-foreground">{userName}</p>
   402→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
   403→          </div>
   404→
   405→          {/* Payment Proof Details */}
   406→          <div className="p-4 sm:p-5 space-y-4">
   407→            {/* Payment Method */}
   408→            {deal.paymentMethod && (
   409→              <div className="flex items-center gap-3">
   410→                <div className="h-10 w-10 rounded-xl bg-muted/60 flex items-center justify-center shrink-0 overflow-hidden">
   411→                  {deal.paymentMethod.image ? (
   412→                    <img
   413→                      src={deal.paymentMethod.image}
   414→                      alt={deal.paymentMethod.name}
   415→                      className="h-full w-full object-cover"
   416→                      onError={(e) => {
   417→                        (e.target as HTMLImageElement).style.display = 'none';
   418→                        (e.target as HTMLImageElement).nextElementSibling?.classList.remove('hidden');
   419→                      }}
   420→                    />
   421→                  ) : null}
   422→                  <span className={`text-xs font-bold ${deal.paymentMethod.image ? 'hidden' : ''}`}>
   423→                    {deal.paymentMethod.name.charAt(0)}
   424→                  </span>
   425→                </div>
   426→                <div className="min-w-0">
   427→                  <p className="text-xs text-muted-foreground">পেমেন্ট মেথড</p>
   428→                  <p className="text-sm font-semibold text-foreground">
   429→                    {deal.paymentMethod.name}
   430→                    <span className="text-muted-foreground font-normal"> ({deal.paymentMethod.accountType === 'merchant' ? 'মার্চেন্ট' : 'পার্সোনাল'})</span>
   431→                  </p>
   432→                </div>
   433→              </div>
   434→            )}
   435→
   436→            {/* Account Number (senderNumber) */}
   437→            <div className="rounded-xl border border-border/50 bg-muted/20 p-3.5">
   438→              <div className="flex items-center gap-2 mb-1.5">
   439→                <Copy className="h-3.5 w-3.5 text-muted-foreground" />
   440→                <span className="text-xs font-medium text-muted-foreground">সেন্ডার একাউন্ট নম্বর</span>
   441→              </div>
   442→              <p className="text-lg font-bold font-mono text-foreground tracking-wide break-all">

   444→              </p>
   445→            </div>
   446→
   447→            {/* Transaction ID */}
   448→            <div className="rounded-xl border border-border/50 bg-muted/20 p-3.5">
   449→              <div className="flex items-center gap-2 mb-1.5">

   478→            {/* Action Buttons */}
   479→            <div className="flex gap-2 pt-2">
   480→              <Button
   481→                size="lg"

   486→                {isActing ? (
   487→                  <Loader2 className="h-4 w-4 animate-spin" />

   491→                পেমেন্ট ভেরিফাই করুন

   495→                variant="outline"
   496→                disabled={isActing}

   498→                className="flex-1 h-12 gap-2 rounded-xl text-sm font-semibold border-red-200 text-red-600 hover:bg-red-50 dark:border-red-800/50 dark:text-red-400 dark:hover:bg-red-500/10"

   515→  return (

   526→          </p>
