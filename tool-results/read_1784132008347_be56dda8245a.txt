    21→/* In-memory cache */
    22→let _settingsCache: EmailSettings | null = null;
    23→let _settingsLoadedAt = 0;
    24→const SETTINGS_TTL = 5 * 60 * 1000; // 5 min
    25→
    26→/** Load email settings from DB (call before sending) */
    27→export async function loadEmailSettings(): Promise<EmailSettings> {
    28→  const now = Date.now();
    29→  if (_settingsCache && now - _settingsLoadedAt < SETTINGS_TTL) return _settingsCache;
    30→  try {
    31→    const { db } = await import('@/lib/db');
    32→    const rows = await db.platformSetting.findMany({
    33→      where: { key: { in: EMAIL_SETTING_KEYS } },
    34→    });
    35→    const overrides: EmailSettings = {};
    36→    for (const r of rows) overrides[r.key] = r.value;
    37→    _settingsCache = { ...DEFAULTS, ...overrides };
    38→    _settingsLoadedAt = now;
    39→  } catch (e) {
    40→    console.error('[EMAIL SETTINGS] DB load failed, using defaults', e);
    41→    _settingsCache = { ...DEFAULTS };
    42→    _settingsLoadedAt = now;
    43→  }
    44→  return _settingsCache;
    45→}

   576→function otpDigitsHtml(otp: string): string {

   621→type EmailPayload = { subject: string; html: string };
   622→
   623→export async function sendEmail(to: string, payload: EmailPayload): Promise<void> {
   624→  const settings = await loadEmailSettings(); // refresh cache if stale
   625→  const transporter = getTransporter(settings);
   626→  if (!transporter) throw new Error('BREVO_SMTP_KEY সেট করা নেই। .env ফাইলে বা অ্যাডমিন প্যানেলে যোগ করুন।');
   627→
   628→  await transporter.sendMail({
   629→    from: `${s('email_from_name')} <${getFromAddress(settings)}>`,
   630→    to,
   631→    subject: payload.subject,
   632→    html: payload.html,
   633→  });
   634→  console.log(`[EMAIL SENT] → ${to}: ${payload.subject}`);
   635→}
   636→
   637→export function fireEmails(emails: Array<{ to: string; payload: EmailPayload }>) {
   638→  for (const e of emails) {
   639→    sendEmail(e.to, e.payload).catch((err) => {
   640→      console.error(`[EMAIL ERROR] → ${e.to}:`, err);
   641→    });
   642→  }
