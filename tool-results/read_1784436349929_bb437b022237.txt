    74→function adminChatBubble(msg: AdminChatMsg) {

    90→  const labelStyle = isSystem
    91→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    92→    : isAdmin
    93→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    94→    : isBuyer
    95→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'

    98→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';

   444→function PaymentVerifyPanel() {
   445→  const [deals, setDeals] = useState<DealRow[]>([]);
   446→  const [loading, setLoading] = useState(true);
   447→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   448→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   449→  const [confirmDialog, setConfirmDialog] = useState<{
   450→    type: 'verify' | 'wrong_info' | 'cancel';
   451→    deal: DealRow;
   452→  } | null>(null);
   453→  const [editedAmount, setEditedAmount] = useState('');
   454→  const [isUpdatingAmount, setIsUpdatingAmount] = useState(false);
   455→
   456→  // Initialize edited amount when a deal is selected
   457→  useEffect(() => {
   458→    if (selectedDeal) {
   459→      setEditedAmount(String(selectedDeal.paymentAmount || selectedDeal.amount || ''));
   460→    }
   461→  }, [selectedDeal]);
   462→
   463→  const fetchDeals = useCallback(async () => {
   464→    try {
   465→      const res = await fetch('/api/admin/deals');
   466→      if (res.ok) {
   467→        const data = await res.json();
   468→        setDeals(data);
   469→      }
   470→    } catch {
   471→      // use fallback
   472→    } finally {
   473→      setLoading(false);
   474→    }
   475→  }, []);
   476→
   477→  useEffect(() => {
   478→    fetchDeals();
   479→  }, [fetchDeals]);
   480→
   481→  const handleVerify = async () => {
   482→    if (!confirmDialog) return;
   483→    const dealId = confirmDialog.deal.id;
   484→    setActionLoading(dealId);
   485→    try {
   486→      const res = await fetch('/api/admin/verify', {
   487→        method: 'POST',
   488→        headers: { 'Content-Type': 'application/json' },
   489→        body: JSON.stringify({ dealId }),
   490→      });
   491→      if (res.ok) {
   492→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   493→        setConfirmDialog(null);
   494→        setSelectedDeal(null);
   495→        fetchDeals();
   496→      } else {
   497→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   498→        setConfirmDialog(null);
   499→      }
   500→    } catch {
   501→      toast.error('সার্ভারে সমস্যা');
   502→      setConfirmDialog(null);
   503→    } finally {
   504→      setActionLoading(null);
   505→    }
   506→  };
