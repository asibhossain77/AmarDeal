   118→/**
   119→ * Maps a deal status to the active step index.
   120→ * Steps 0-4 correspond to the 5 STEPS.
   121→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
   122→ */
   123→function statusToActiveStep(status: string): number {
   124→  switch (status) {
   125→    case 'created': return 0;
   126→    case 'payment_pending': return 1;
   127→    case 'payment_verified': return 2;
   128→    case 'in_delivery': return 3;
   129→    case 'completed': return 4;
   130→    case 'cancelled':
   131→    case 'rejected': return -1;
   132→    case 'disputed': return 3;
   133→    default: return 0;
   134→  }
   135→}

   179→interface ChatMessage {

   188→/** Convert ISO date string to Bengali time */
   189→function toBnTime(isoString: string): string {
   190→  try {
   191→    return new Date(isoString).toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
   192→  } catch {
   193→    return '';
   194→  }
   195→}
   196→
   197→/** Convert DB message to ChatMessage */
   198→function dbToChatMsg(m: { id: string; role: string | null; senderName: string | null; text: string; createdAt: string; senderId: string }): ChatMessage {
   199→  return {
   200→    id: m.id,
   201→    role: (m.role as MessageRole) || 'system',
   202→    senderId: m.senderId,
   203→    senderName: m.senderName || 'অজানা',
   204→    text: m.text,
   205→    timestamp: toBnTime(m.createdAt),
   206→  };
   207→}

   241→/**
   242→ * Horizontal 5-step progress bar (desktop).
   243→ * Uses parrot green #65A30D for active/completed steps.
   244→ */
   245→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   246→  if (isCancelled) {
   247→    return (
   248→      <div className="flex items-center justify-center py-6">
   249→        <div className="flex items-center gap-3 rounded-2xl bg-red-50 dark:bg-red-500/10 px-6 py-4 border border-red-200 dark:border-red-500/20">
   250→          <Ban className="h-8 w-8 text-red-500" />
   251→          <div>
   252→            <p className="text-base font-bold text-red-700 dark:text-red-400">ডিল বাতিল হয়েছে</p>
   253→            <p className="text-sm text-red-600/70 dark:text-red-400/70">এই ডিল আর সক্রিয় নয়</p>
   254→          </div>
   255→        </div>
   256→      </div>
   257→    );
   258→  }
   259→
   260→  return (
   261→    <div className="relative flex items-start justify-between py-2">
   262→      {/* Background track line */}
   263→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />
   264→      {/* Filled progress line */}
   265→      {activeStep >= 0 && (
   266→        <div
   267→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   268→          style={{
   269→            width: activeStep >= STEPS.length - 1
   270→              ? 'calc(100% - 40px)'
   271→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   272→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   273→          }}
   274→        />
   275→      )}
   276→
   277→      {STEPS.map((step, i) => {
   278→        const Icon = step.icon;
   279→        const isCompleted = i < activeStep;
   280→        const isActive = i === activeStep;
   281→
   282→        return (
   283→          <div
   284→            key={step.num}
   285→            className="relative z-10 flex flex-col items-center"
   286→            style={{ width: `${100 / STEPS.length}%` }}
   287→          >
   288→            <div className="relative">
   289→              {/* Pulse rings for active step */}
   290→              {isActive && (
   291→                <>
   292→                  <span
   293→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   294→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   295→                  />
   296→                  <span
   297→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   298→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   299→                  />
   300→                </>
   301→              )}
   302→              <div
   303→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   304→                style={{
   305→                  backgroundColor: isCompleted || isActive
   306→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)
   307→                    : 'var(--background)',
   308→                  borderColor: isCompleted || isActive
   309→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)
   310→                    : 'var(--border)',
   311→                  color: isCompleted ? '#fff' : (isActive && isDisputed ? '#EF4444' : (isActive ? '#fff' : 'var(--muted-foreground)')),
   312→                  boxShadow: (isCompleted || isActive) && !isDisputed ? PARROT_GREEN_GLOW : 'none',
   313→                }}
   314→              >
   315→                {isCompleted ? (
   316→                  <Check className="h-4 w-4" strokeWidth={3} />
   317→                ) : isActive && isDisputed ? (
   318→                  <AlertTriangle className="h-4 w-4" />
   319→                ) : (
   320→                  <Icon className="h-4 w-4" />
   321→                )}
   322→              </div>
   323→            </div>
   324→            <p
   325→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   326→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   327→            >
   328→              {step.label}
   329→            </p>
   330→          </div>
   331→        );
   332→      })}
   333→    </div>
   334→  );
   335→}
