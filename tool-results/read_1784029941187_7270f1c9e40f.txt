    69→/* ─── Admin Chat Message Styles ─── */
    70→function adminChatBubble(msg: AdminChatMsg) {
    71→  const isAdmin = msg.role === 'admin';
    72→  const isSystem = msg.role === 'system';
    73→  const isBuyer = msg.role === 'buyer';
    74→  const isSeller = msg.role === 'seller';
    75→
    76→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    77→
    78→  const bubbleStyle = isSystem
    79→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    80→    : isAdmin
    81→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    82→    : isBuyer
    83→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    84→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    85→
    86→  const labelStyle = isSystem
    87→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    88→    : isAdmin
    89→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    90→    : isBuyer
    91→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    92→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    93→
    94→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    95→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    96→
    97→  return (
    98→    <div key={msg.id} className={`flex ${align}`}>
    99→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
   100→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
   101→        <p className="leading-relaxed">{msg.text}</p>
   102→      </div>
   103→    </div>
   104→  );
   105→}

   440→function PaymentVerifyPanel() {
   441→  const [deals, setDeals] = useState<DealRow[]>([]);
   442→  const [loading, setLoading] = useState(true);
   443→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   444→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   445→  const [confirmDialog, setConfirmDialog] = useState<{
   446→    type: 'verify' | 'wrong_info' | 'cancel';
   447→    deal: DealRow;
   448→  } | null>(null);
   449→  const [editedAmount, setEditedAmount] = useState('');
   450→  const [isUpdatingAmount, setIsUpdatingAmount] = useState(false);
   451→
   452→  // Initialize edited amount when a deal is selected
   453→  useEffect(() => {
   454→    if (selectedDeal) {
   455→      setEditedAmount(String(selectedDeal.paymentAmount || selectedDeal.amount || ''));
   456→    }
   457→  }, [selectedDeal]);
   458→
   459→  const fetchDeals = useCallback(async () => {
   460→    try {
   461→      const res = await fetch('/api/admin/deals');
   462→      if (res.ok) {
   463→        const data = await res.json();
   464→        setDeals(data);
   465→      }
   466→    } catch {
   467→      // use fallback
   468→    } finally {
   469→      setLoading(false);
   470→    }
   471→  }, []);
   472→
   473→  useEffect(() => {
   474→    fetchDeals();
   475→  }, [fetchDeals]);
   476→
   477→  const handleVerify = async () => {
   478→    if (!confirmDialog) return;
   479→    const dealId = confirmDialog.deal.id;
   480→    setActionLoading(dealId);
   481→    try {
   482→      const res = await fetch('/api/admin/verify', {
   483→        method: 'POST',
   484→        headers: { 'Content-Type': 'application/json' },
   485→        body: JSON.stringify({ dealId }),
   486→      });
   487→      if (res.ok) {
   488→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   489→        setConfirmDialog(null);
   490→        setSelectedDeal(null);
   491→        fetchDeals();
   492→      } else {
   493→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   494→        setConfirmDialog(null);
   495→      }
   496→    } catch {
   497→      toast.error('সার্ভারে সমস্যা');
   498→      setConfirmDialog(null);
   499→    } finally {
   500→      setActionLoading(null);
   501→    }
   502→  };
   503→
   504→  const handleWrongInfo = async () => {
   505→    if (!confirmDialog) return;
   506→    const dealId = confirmDialog.deal.id;
   507→    setActionLoading(dealId);
   508→    try {
   509→      const res = await fetch('/api/admin/reject', {
   510→        method: 'POST',
   511→        headers: { 'Content-Type': 'application/json' },
   512→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   513→      });
   514→      if (res.ok) {
   515→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   516→        setConfirmDialog(null);
   517→        setSelectedDeal(null);
   518→        fetchDeals();
   519→      } else {
   520→        const data = await res.json().catch(() => ({}));
   521→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   522→        setConfirmDialog(null);
   523→      }
   524→    } catch {
   525→      toast.error('সার্ভারে সমস্যা');
   526→      setConfirmDialog(null);
   527→    } finally {
   528→      setActionLoading(null);
   529→    }
   530→  };
   531→
   532→  const handleCancelDeal = async () => {
   533→    if (!confirmDialog) return;
   534→    const dealId = confirmDialog.deal.id;
   535→    setActionLoading(dealId);
   536→    try {
   537→      const res = await fetch('/api/admin/reject', {
   538→        method: 'POST',
   539→        headers: { 'Content-Type': 'application/json' },
   540→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   541→      });
   542→      if (res.ok) {
