    59→/* ─── Known Gateway Defaults ─── */
    60→function getGatewayColor(name: string): string {
    61→  const n = name.toLowerCase();
    62→  if (n.includes('bkash')) return '#E2136E';
    63→  if (n.includes('nagad')) return '#F6921E';
    64→  if (n.includes('rocket')) return '#8C3494';
    65→  if (n.includes('bank')) return '#0033A0';
    66→  if (n.includes('upay')) return '#1A9F4B';
    67→  if (n.includes('dutch') || n.includes('bangla')) return '#FF6600';
    68→  return '#84CC16';
    69→}
    70→
    71→function getGatewayIcon(name: string) {
    72→  const n = name.toLowerCase();
    73→  if (n.includes('bank')) return '🏦';
    74→  return '💳';
    75→}

    94→/* ─── Color Picker Row ─── */
    95→function ColorPickerRow({
    96→  color,
    97→  onChange,
    98→  label,
    99→}: {
   100→  color: string;
   101→  onChange: (c: string) => void;
   102→  label?: string;
   103→}) {
   104→  return (
   105→    <div className="space-y-1.5">
   106→      {label && <Label className="text-sm font-medium">{label}</Label>}
   107→      <div className="flex items-center gap-3">
   108→        <div className="relative">
   109→          <input
   110→            type="color"
   111→            value={color}
   112→            onChange={(e) => onChange(e.target.value)}
   113→            className="h-10 w-14 cursor-pointer rounded-lg border border-border p-1 bg-transparent"
   114→          />
   115→        </div>
   116→        <Input
   117→          value={color}
   118→          onChange={(e) => {
   119→            const v = e.target.value;
   120→            if (/^#[0-9a-fA-F]{0,6}$/.test(v)) {
   121→              onChange(v.length === 7 ? v : color);
   122→            }
   123→          }}
   124→          className="w-32 h-10 font-mono text-sm"
   125→        />
   126→        <div
   127→          className="h-10 w-20 rounded-xl border border-border/50 shrink-0"
   128→          style={{ backgroundColor: color }}
   129→        />
   130→      </div>
   131→      {/* Quick color presets */}
   132→      <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   133→        {['#E2136E', '#F6921E', '#8C3494', '#0033A0', '#1A9F4B', '#84CC16', '#65A30D', '#2563EB', '#DC2626', '#7C3AED'].map((c) => (
   134→          <button
   135→            key={c}
   136→            type="button"
   137→            onClick={() => onChange(c)}
   138→            className="h-6 w-6 rounded-full border-2 transition-transform hover:scale-110"
   139→            style={{
   140→              backgroundColor: c,
   141→              borderColor: color === c ? 'var(--foreground)' : 'transparent',
   142→            }}
   143→            title={c}
   144→          />
   145→        ))}
   146→      </div>
   147→    </div>
   148→  );
   149→}

   762→function getContrastColor(hex: string): string {
   763→  const r = parseInt(hex.slice(1, 3), 16);
   764→  const g = parseInt(hex.slice(3, 5), 16);
   765→  const b = parseInt(hex.slice(5, 7), 16);
   766→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   767→  return luminance > 0.6 ? '#18181b' : '#ffffff';
   768→}
