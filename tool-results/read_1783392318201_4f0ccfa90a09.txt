    63→/**
    64→ * Maps a deal status to the active step index.
    65→ * Steps 0-4 correspond to the 5 STEPS.
    66→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
    67→ */
    68→function statusToActiveStep(status: string): number {
    69→  switch (status) {
    70→    case 'created': return 0;
    71→    case 'payment_pending': return 1;
    72→    case 'payment_verified': return 2;
    73→    case 'in_delivery': return 3;
    74→    case 'completed': return 4;
    75→    case 'cancelled':
    76→    case 'rejected': return -1;
    77→    case 'disputed': return 3;
    78→    default: return 0;
    79→  }
    80→}

   132→/** Convert ISO date string to Bengali time */
   133→function toBnTime(isoString: string): string {
   134→  try {
   135→    return new Date(isoString).toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
   136→  } catch {
   137→    return '';
   138→  }
   139→}

   141→/** Convert DB message to ChatMessage */
   142→function dbToChatMsg(m: { id: string; role: string | null; senderName: string | null; text: string; createdAt: string; senderId: string }): ChatMessage {
   143→  return {
   144→    id: m.id,
   145→    role: (m.role as MessageRole) || 'buyer',
   146→    senderName: m.senderName || 'অজানা',
   147→    text: m.text,
   148→    timestamp: toBnTime(m.createdAt),

   180→/**
   181→ * Horizontal 5-step progress bar (desktop).
   182→ * Uses parrot green #65A30D for active/completed steps.
   183→ */
   184→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   185→  if (isCancelled) {
   186→    return (
   187→      <div className="flex items-center justify-center py-6">
   188→        <div className="flex items-center gap-3 rounded-2xl bg-red-50 dark:bg-red-500/10 px-6 py-4 border border-red-200 dark:border-red-500/20">
   189→          <Ban className="h-8 w-8 text-red-500" />
   190→          <div>
   191→            <p className="text-base font-bold text-red-700 dark:text-red-400">ডিল বাতিল হয়েছে</p>
   192→            <p className="text-sm text-red-600/70 dark:text-red-400/70">এই ডিল আর সক্রিয় নয়</p>
   193→          </div>
   194→        </div>
   195→      </div>
   196→    );
   197→  }
   198→
   199→  return (
   200→    <div className="relative flex items-start justify-between py-2">
   201→      {/* Background track line */}
   202→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />
   203→      {/* Filled progress line */}
   204→      {activeStep >= 0 && (
   205→        <div
   206→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   207→          style={{
   208→            width: activeStep >= STEPS.length - 1
   209→              ? 'calc(100% - 40px)'
   210→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   211→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   212→          }}
   213→        />
   214→      )}
   215→
   216→      {STEPS.map((step, i) => {
   217→        const Icon = step.icon;
   218→        const isCompleted = i < activeStep;
   219→        const isActive = i === activeStep;
   220→
   221→        return (
   222→          <div
   223→            key={step.num}
   224→            className="relative z-10 flex flex-col items-center"
   225→            style={{ width: `${100 / STEPS.length}%` }}
   226→          >
   227→            <div className="relative">
   228→              {/* Pulse rings for active step */}
   229→              {isActive && (
   230→                <>
   231→                  <span
   232→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   233→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   234→                  />
   235→                  <span
   236→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   237→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   238→                  />
   239→                </>
   240→              )}
   241→              <div
   242→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   243→                style={{
   244→                  backgroundColor: isCompleted || isActive
   245→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)
   246→                    : 'var(--background)',
   247→                  borderColor: isCompleted || isActive
   248→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)
   249→                    : 'var(--border)',
   250→                  color: isCompleted ? '#fff' : (isActive && isDisputed ? '#EF4444' : (isActive ? '#fff' : 'var(--muted-foreground)')),
   251→                  boxShadow: (isCompleted || isActive) && !isDisputed ? PARROT_GREEN_GLOW : 'none',
   252→                }}
   253→              >
   254→                {isCompleted ? (
   255→                  <Check className="h-4 w-4" strokeWidth={3} />
   256→                ) : isActive && isDisputed ? (
   257→                  <AlertTriangle className="h-4 w-4" />
   258→                ) : (
   259→                  <Icon className="h-4 w-4" />
   260→                )}
   261→              </div>
   262→            </div>
   263→            <p
   264→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   265→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   266→            >
   267→              {step.label}
   268→            </p>
   269→          </div>
   270→        );
   271→      })}
   272→    </div>
   273→  );
   274→}
