    65→/* ─── Admin Chat Message Styles ─── */
    66→function adminChatBubble(msg: AdminChatMsg) {
    67→  const isAdmin = msg.role === 'admin';
    68→  const isSystem = msg.role === 'system';
    69→  const isBuyer = msg.role === 'buyer';
    70→  const isSeller = msg.role === 'seller';
    71→
    72→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    73→
    74→  const bubbleStyle = isSystem
    75→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    76→    : isAdmin
    77→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    78→    : isBuyer
    79→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    80→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    81→
    82→  const labelStyle = isSystem
    83→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    84→    : isAdmin
    85→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    86→    : isBuyer
    87→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    88→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    89→
    90→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    91→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    92→
    93→  return (
    94→    <div key={msg.id} className={`flex ${align}`}>
    95→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>

   400→function PaymentVerifyPanel() {
   401→  const [deals, setDeals] = useState<DealRow[]>([]);
   402→  const [loading, setLoading] = useState(true);
   403→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   404→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   405→  const [confirmDialog, setConfirmDialog] = useState<{
   406→    type: 'verify' | 'wrong_info' | 'cancel';
   407→    deal: DealRow;
   408→  } | null>(null);
   409→
   410→  const fetchDeals = useCallback(async () => {
   411→    try {
   412→      const res = await fetch('/api/admin/deals');
   413→      if (res.ok) {
   414→        const data = await res.json();
   415→        setDeals(data);
   416→      }
   417→    } catch {
   418→      // use fallback
   419→    } finally {
   420→      setLoading(false);
   421→    }
   422→  }, []);
   423→
   424→  useEffect(() => {
   425→    fetchDeals();
   426→  }, [fetchDeals]);
   427→
   428→  const handleVerify = async () => {
   429→    if (!confirmDialog) return;
   430→    const dealId = confirmDialog.deal.id;
   431→    setActionLoading(dealId);
   432→    try {
   433→      const res = await fetch('/api/admin/verify', {
   434→        method: 'POST',
   435→        headers: { 'Content-Type': 'application/json' },
   436→        body: JSON.stringify({ dealId }),
   437→      });
   438→      if (res.ok) {
   439→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   440→        setConfirmDialog(null);
   441→        setSelectedDeal(null);
   442→        fetchDeals();
   443→      } else {
   444→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   445→        setConfirmDialog(null);
   446→      }
   447→    } catch {
   448→      toast.error('সার্ভারে সমস্যা');
   449→      setConfirmDialog(null);
   450→    } finally {
   451→      setActionLoading(null);
   452→    }
   453→  };
   454→
   455→  const handleWrongInfo = async () => {
   456→    if (!confirmDialog) return;
   457→    const dealId = confirmDialog.deal.id;
   458→    setActionLoading(dealId);
   459→    try {
   460→      const res = await fetch('/api/admin/reject', {
   461→        method: 'POST',
   462→        headers: { 'Content-Type': 'application/json' },
   463→        body: JSON.stringify({ dealId, reason: 'wrong_info' }),
   464→      });
   465→      if (res.ok) {
   466→        toast.success('ভুল তথ্য হিসেবে চিহ্নিত করা হয়েছে। ক্রেতা আবার পেমেন্ট করতে পারবে।');
   467→        setConfirmDialog(null);
   468→        setSelectedDeal(null);
   469→        fetchDeals();
   470→      } else {
   471→        const data = await res.json().catch(() => ({}));
   472→        toast.error(data.error || 'ব্যর্থ হয়েছে');
   473→        setConfirmDialog(null);
   474→      }
   475→    } catch {
   476→      toast.error('সার্ভারে সমস্যা');
   477→      setConfirmDialog(null);
   478→    } finally {
   479→      setActionLoading(null);
   480→    }
   481→  };
   482→
   483→  const handleCancelDeal = async () => {
   484→    if (!confirmDialog) return;
   485→    const dealId = confirmDialog.deal.id;
   486→    setActionLoading(dealId);
   487→    try {
   488→      const res = await fetch('/api/admin/reject', {
   489→        method: 'POST',
   490→        headers: { 'Content-Type': 'application/json' },
   491→        body: JSON.stringify({ dealId, reason: 'cancel' }),
   492→      });
   493→      if (res.ok) {
   494→        toast.success('ডিল বাতিল করা হয়েছে');
   495→        setConfirmDialog(null);
   496→        setSelectedDeal(null);
   497→        fetchDeals();
   498→      } else {
   499→        const data = await res.json().catch(() => ({}));
   500→        toast.error(data.error || 'বাতিল করতে সমস্যা হয়েছে');
   501→        setConfirmDialog(null);
   502→      }
   503→    } catch {
   504→      toast.error('সার্ভারে সমস্যা');
   505→      setConfirmDialog(null);
   506→    } finally {
   507→      setActionLoading(null);
   508→    }
   509→  };
   510→
   511→  const pendingDeals = deals.filter(
   512→    (d) =>
   513→      d.status === 'created' ||
   514→      d.status === 'pending' ||
   515→      d.status === 'payment_pending'
   516→  );
   517→
   518→  /* ─── Detail View (shown when a deal is selected) ─── */
   519→  if (selectedDeal) {
   520→    const deal = selectedDeal;
   521→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   522→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   523→    const isActing = actionLoading === deal.id;
   524→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   525→      year: 'numeric',
   526→      month: 'short',
   527→      day: 'numeric',
   528→      hour: '2-digit',
   529→      minute: '2-digit',
   530→    });
   531→
   532→    return (
   533→      <div>
   534→        {/* Back button + header */}
   535→        <div className="mb-4 flex items-center gap-2">
   536→          <button
   537→            onClick={() => setSelectedDeal(null)}
   538→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   539→          >
   540→            <ArrowLeft className="h-4 w-4" />
   541→            ফিরে যান
   542→          </button>
   543→        </div>
   544→
   545→        <SolidCard className="!p-0 overflow-hidden">
   546→          {/* Header */}
   547→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
