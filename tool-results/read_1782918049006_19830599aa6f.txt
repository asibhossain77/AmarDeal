    48→/**
    49→ * Maps a deal status to the active step index.
    50→ * Steps 0-4 correspond to the 5 STEPS.
    51→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
    52→ */
    53→function statusToActiveStep(status: string): number {
    54→  switch (status) {
    55→    case 'created':
    56→      return 0;
    57→    case 'payment_pending':
    58→      return 1;
    59→    case 'payment_verified':
    60→      return 2;
    61→    case 'in_delivery':
    62→      return 3;
    63→    case 'completed':
    64→      return 4;
    65→    case 'cancelled':
    66→    case 'rejected':
    67→      return -1;
    68→    case 'disputed':
    69→      return 3; // Freeze at step 4 with dispute indicator
    70→    default:
    71→      return 0;
    72→  }
    73→}
