   117→/**
   118→ * Maps a deal status to the active step index.
   119→ * Steps 0-4 correspond to the 5 STEPS.
   120→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).
   121→ */
   122→function statusToActiveStep(status: string): number {
   123→  switch (status) {
   124→    case 'created': return 0;
   125→    case 'payment_pending': return 1;
   126→    case 'payment_verified': return 2;
   127→    case 'in_delivery': return 3;
   128→    case 'completed': return 4;
   129→    case 'cancelled':
   130→    case 'rejected': return -1;
   131→    case 'disputed': return 3;
   132→    default: return 0;
   133→  }
   134→}

   178→interface ChatMessage {

   186→/** Convert ISO date string to Bengali time */
   187→function toBnTime(isoString: string): string {
   188→  try {
   189→    return new Date(isoString).toLocaleTimeString('bn-BD', { hour: '2-digit', minute: '2-digit' });
   190→  } catch {
   191→    return '';
   192→  }
   193→}
   194→
   195→/** Convert DB message to ChatMessage */
   196→function dbToChatMsg(m: { id: string; role: string | null; senderName: string | null; text: string; createdAt: string; senderId: string }): ChatMessage {
   197→  return {
   198→    id: m.id,
   199→    role: (m.role as MessageRole) || 'buyer',
   200→    senderName: m.senderName || 'অজানা',
   201→    text: m.text,
   202→    timestamp: toBnTime(m.createdAt),
   203→  };

   237→/**
   238→ * Horizontal 5-step progress bar (desktop).
   239→ * Uses parrot green #65A30D for active/completed steps.
   240→ */
   241→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   242→  if (isCancelled) {
   243→    return (
   244→      <div className="flex items-center justify-center py-6">
   245→        <div className="flex items-center gap-3 rounded-2xl bg-red-50 dark:bg-red-500/10 px-6 py-4 border border-red-200 dark:border-red-500/20">

   247→          <div>
   248→            <p className="text-base font-bold text-red-700 dark:text-red-400">ডিল বাতিল হয়েছে</p>
   249→            <p className="text-sm text-red-600/70 dark:text-red-400/70">এই ডিল আর সক্রিয় নয়</p>
   250→          </div>
   251→        </div>
   252→      </div>
   253→    );
   254→  }
   255→
   256→  return (
   257→    <div className="relative flex items-start justify-between py-2">
   258→      {/* Background track line */}
   259→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />
   260→      {/* Filled progress line */}
   261→      {activeStep >= 0 && (
   262→        <div
   263→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   264→          style={{
   265→            width: activeStep >= STEPS.length - 1
   266→              ? 'calc(100% - 40px)'
   267→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   268→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   269→          }}
   270→        />
   271→      )}
   272→
   273→      {STEPS.map((step, i) => {
   274→        const Icon = step.icon;
   275→        const isCompleted = i < activeStep;
   276→        const isActive = i === activeStep;
   277→
   278→        return (
   279→          <div
   280→            key={step.num}
   281→            className="relative z-10 flex flex-col items-center"
   282→            style={{ width: `${100 / STEPS.length}%` }}
   283→          >
   284→            <div className="relative">
   285→              {/* Pulse rings for active step */}
   286→              {isActive && (
   287→                <>
   288→                  <span
   289→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   290→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   291→                  />
   292→                  <span
   293→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   294→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   295→                  />
   296→                </>
   297→              )}
   298→              <div
   299→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   300→                style={{
   301→                  backgroundColor: isCompleted || isActive
   302→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)
   303→                    : 'var(--background)',
   304→                  borderColor: isCompleted || isActive
   305→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)
   306→                    : 'var(--border)',
   307→                  color: isCompleted ? '#fff' : (isActive && isDisputed ? '#EF4444' : (isActive ? '#fff' : 'var(--muted-foreground)')),
   308→                  boxShadow: (isCompleted || isActive) && !isDisputed ? PARROT_GREEN_GLOW : 'none',
   309→                }}
   310→              >
   311→                {isCompleted ? (
   312→                  <Check className="h-4 w-4" strokeWidth={3} />
   313→                ) : isActive && isDisputed ? (
   314→                  <AlertTriangle className="h-4 w-4" />
   315→                ) : (
   316→                  <Icon className="h-4 w-4" />
   317→                )}
   318→              </div>
   319→            </div>
   320→            <p
   321→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   322→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   323→            >
   324→              {step.label}
   325→            </p>
   326→          </div>
   327→        );
   328→      })}
   329→    </div>
   330→  );
   331→}
