    69→    case 'created': return 0;
    70→    case 'payment_pending': return 1;
    71→    case 'payment_verified': return 2;
    72→    case 'in_delivery': return 3;
    73→    case 'completed': return 4;
    74→    case 'cancelled':
    75→    case 'rejected': return -1;
    76→    case 'disputed': return 3;
    77→    default: return 0;
    78→  }
    79→}

   210→            width: activeStep >= STEPS.length - 1
   211→              ? 'calc(100% - 40px)'

   213→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,

   217→

   220→        const isCompleted = i < activeStep;

   269→              {step.label}

   273→      })}
