    65→/**

    72→    case 'created': return 0;
    73→    case 'payment_pending': return 1;

    75→    case 'in_delivery': return 3;
    76→    case 'completed': return 4;
    77→    case 'cancelled':
    78→    case 'rejected': return -1;
    79→    case 'disputed': return 3;
    80→    default: return 0;
    81→  }

   190→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {

   212→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   213→          style={{
   214→            width: activeStep >= STEPS.length - 1
   215→              ? 'calc(100% - 40px)'
   216→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   217→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   218→          }}
   219→        />
   220→      )}
   221→

   223→        const Icon = step.icon;
   224→        const isCompleted = i < activeStep;
   225→        const isActive = i === activeStep;

   228→          <div

   241→                  <span
   242→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"

   248→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"

   273→              {step.label}
   274→            </p>
   275→          </div>
   276→        );
   277→      })}
   278→    </div>
   279→  );
