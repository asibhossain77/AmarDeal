   308→
   309→function PaymentVerifyPanel() {
   310→  const [deals, setDeals] = useState<DealRow[]>([]);
   311→  const [loading, setLoading] = useState(true);
   312→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   313→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   314→  const [confirmDialog, setConfirmDialog] = useState<{
   315→    type: 'verify' | 'reject';
   316→    deal: DealRow;
   317→  } | null>(null);
   318→
   319→  const fetchDeals = useCallback(async () => {
   320→    try {
   321→      const res = await fetch('/api/admin/deals');
   322→      if (res.ok) {
   323→        const data = await res.json();
   324→        setDeals(data);
   325→      }
   326→    } catch {
   327→      // use fallback
   328→    } finally {
   329→      setLoading(false);
   330→    }
   331→  }, []);
   332→
   333→  useEffect(() => {
   334→    fetchDeals();
   335→  }, [fetchDeals]);
   336→
   337→  const handleVerify = async () => {
   338→    if (!confirmDialog) return;
   339→    const dealId = confirmDialog.deal.id;
   340→    setActionLoading(dealId);
   341→    try {
   342→      const res = await fetch('/api/admin/verify', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({ dealId }),
   346→      });
   347→      if (res.ok) {
   348→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   349→        setConfirmDialog(null);
   350→        setSelectedDeal(null);
   351→        fetchDeals();
   352→      } else {
   353→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   354→        setConfirmDialog(null);
   355→      }
   356→    } catch {
   357→      toast.error('সার্ভারে সমস্যা');
   358→      setConfirmDialog(null);
   359→    } finally {
   360→      setActionLoading(null);
   361→    }
   362→  };
   363→
   364→  const handleReject = async () => {
   365→    if (!confirmDialog) return;
   366→    const dealId = confirmDialog.deal.id;
   367→    setActionLoading(dealId);
   368→    try {
   369→      const res = await fetch('/api/admin/reject', {
   370→        method: 'POST',
   371→        headers: { 'Content-Type': 'application/json' },
   372→        body: JSON.stringify({ dealId }),
   373→      });
   374→      if (res.ok) {
   375→        toast.success('ডিল রিজেক্ট করা হয়েছে');
   376→        setConfirmDialog(null);
   377→        setSelectedDeal(null);
   378→        fetchDeals();
   379→      } else {
   380→        toast.error('রিজেক্ট ব্যর্থ হয়েছে');
   381→        setConfirmDialog(null);
   382→      }
   383→    } catch {
   384→      toast.error('সার্ভারে সমস্যা');
   385→      setConfirmDialog(null);
   386→    } finally {
   387→      setActionLoading(null);
   388→    }
   389→  };
   390→
   391→  const pendingDeals = deals.filter(
   392→    (d) =>
   393→      d.status === 'created' ||
   394→      d.status === 'pending' ||
   395→      d.status === 'payment_pending'
   396→  );
   397→
   398→  /* ─── Detail View (shown when a deal is selected) ─── */
   399→  if (selectedDeal) {
   400→    const deal = selectedDeal;
   401→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   402→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   403→    const isActing = actionLoading === deal.id;
   404→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   405→      year: 'numeric',
   406→      month: 'short',
   407→      day: 'numeric',
   408→      hour: '2-digit',
   409→      minute: '2-digit',
   410→    });
   411→
   412→    return (
   413→      <div>
   414→        {/* Back button + header */}
   415→        <div className="mb-4 flex items-center gap-2">
   416→          <button
   417→            onClick={() => setSelectedDeal(null)}
   418→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   419→          >
   420→            <ArrowLeft className="h-4 w-4" />
   421→            ফিরে যান
   422→          </button>
   423→        </div>
   424→
   425→        <SolidCard className="!p-0 overflow-hidden">
   426→          {/* Header */}
   427→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   428→            <div className="flex items-center justify-between mb-2">
   429→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   430→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   431→                পেন্ডিং
   432→              </Badge>
   433→            </div>
   434→            <p className="text-base font-bold text-foreground">{userName}</p>
   435→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
   436→          </div>
   437→
   438→          {/* Payment Proof Details */}
   439→          <div className="p-4 sm:p-5 space-y-4">
   440→            {/* Payment Method */}
   441→            {deal.paymentMethod && (
   442→              <div className="flex items-center gap-3">
   443→                <div className="h-10 w-10 rounded-xl bg-muted/60 flex items-center justify-center shrink-0 overflow-hidden">
   444→                  {deal.paymentMethod.image ? (
   445→                    <img
   446→                      src={deal.paymentMethod.image}

   451→                        (e.target as HTMLImageElement).nextElementSibling?.classList.remove('hidden');
   452→                      }}
   453→                    />
   454→                  ) : null}
   455→                  <span className={`text-xs font-bold ${deal.paymentMethod.image ? 'hidden' : ''}`}>
