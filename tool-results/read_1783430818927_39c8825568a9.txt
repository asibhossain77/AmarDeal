   301→function PaymentVerifyPanel() {
   302→  const [deals, setDeals] = useState<DealRow[]>([]);
   303→  const [loading, setLoading] = useState(true);
   304→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   305→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   306→
   307→  const fetchDeals = useCallback(async () => {
   308→    try {
   309→      const res = await fetch('/api/admin/deals');
   310→      if (res.ok) {
   311→        const data = await res.json();
   312→        setDeals(data);
   313→      }
   314→    } catch {
   315→      // use fallback
   316→    } finally {
   317→      setLoading(false);
   318→    }
   319→  }, []);
   320→
   321→  useEffect(() => {
   322→    fetchDeals();
   323→  }, [fetchDeals]);
   324→
   325→  const handleVerify = async (dealId: string) => {
   326→    setActionLoading(dealId);
   327→    try {
   328→      const res = await fetch('/api/admin/verify', {
   329→        method: 'POST',
   330→        headers: { 'Content-Type': 'application/json' },
   331→        body: JSON.stringify({ dealId }),
   332→      });

   334→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   335→        setSelectedDeal(null);
   336→        fetchDeals();

   338→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   339→      }
   340→    } catch {
   341→      toast.error('সার্ভারে সমস্যা');
   342→    } finally {
   343→      setActionLoading(null);
   344→    }
   345→  };
   346→

   351→        method: 'POST',
   352→        headers: { 'Content-Type': 'application/json' },
   353→        body: JSON.stringify({ dealId }),

   361→      }
   362→    } catch {
   363→      toast.error('সার্ভারে সমস্যা');
   364→    } finally {
   365→      setActionLoading(null);
   366→    }
   367→  };
   368→
   369→  const pendingDeals = deals.filter(
   370→    (d) =>
   371→      d.status === 'created' ||
   372→      d.status === 'pending' ||
   373→      d.status === 'payment_pending'
   374→  );
   375→
   376→  /* ─── Detail View (shown when a deal is selected) ─── */

   378→    const deal = selectedDeal;
   379→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   380→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   381→    const isActing = actionLoading === deal.id;
   382→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   383→      year: 'numeric',
   384→      month: 'short',
   385→      day: 'numeric',
   386→      hour: '2-digit',
   387→      minute: '2-digit',
   388→    });
   389→
   390→    return (
   391→      <div>
   392→        {/* Back button + header */}
   393→        <div className="mb-4 flex items-center gap-2">
   394→          <button
   395→            onClick={() => setSelectedDeal(null)}
   396→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   397→          >
   398→            <ArrowLeft className="h-4 w-4" />
   399→            ফিরে যান
   400→          </button>
   401→        </div>
   402→
   403→        <SolidCard className="!p-0 overflow-hidden">
   404→          {/* Header */}
   405→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   406→            <div className="flex items-center justify-between mb-2">
   407→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   408→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   409→                পেন্ডিং

   411→            </div>

   413→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
