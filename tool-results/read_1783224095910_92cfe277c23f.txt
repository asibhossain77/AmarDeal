    63→/**

    66→ * -1 means no steps are active (cancelled/rejected/disputed before delivery).

    68→function statusToActiveStep(status: string): number {
    69→  switch (status) {
    70→    case 'created': return 0;
    71→    case 'payment_pending': return 1;
    72→    case 'payment_verified': return 2;
    73→    case 'in_delivery': return 3;
    74→    case 'completed': return 4;
    75→    case 'cancelled':
    76→    case 'rejected': return -1;
    77→    case 'disputed': return 3;
    78→    default: return 0;
    79→  }
    80→}

   188→function HorizontalStepper({ activeStep, isDisputed, isCancelled }: { activeStep: number; isDisputed: boolean; isCancelled: boolean }) {
   189→  if (isCancelled) {
   190→    return (
   191→      <div className="flex items-center justify-center py-6">

   197→          </div>
   198→        </div>
   199→      </div>
   200→    );
   201→  }

   203→  return (
   204→    <div className="relative flex items-start justify-between py-2">
   205→      {/* Background track line */}
   206→      <div className="absolute top-[20px] left-[20px] right-[20px] h-[4px] rounded-full bg-border/40" />

   209→        <div
   210→          className="absolute top-[20px] left-[20px] h-[4px] rounded-full transition-all duration-700 ease-out"
   211→          style={{
   212→            width: activeStep >= STEPS.length - 1
   213→              ? 'calc(100% - 40px)'
   214→              : `calc(${(activeStep / (STEPS.length - 1)) * 100}% - 20px)`,
   215→            backgroundColor: isDisputed ? '#EF4444' : PARROT_GREEN,
   216→          }}
   217→        />
   218→      )}
   219→
   220→      {STEPS.map((step, i) => {
   221→        const Icon = step.icon;
   222→        const isCompleted = i < activeStep;
   223→        const isActive = i === activeStep;
   224→
   225→        return (
   226→          <div
   227→            key={step.num}
   228→            className="relative z-10 flex flex-col items-center"
   229→            style={{ width: `${100 / STEPS.length}%` }}

   231→            <div className="relative">
   232→              {/* Pulse rings for active step */}
   233→              {isActive && (
   234→                <>
   235→                  <span
   236→                    className="absolute -inset-2.5 rounded-full border-2 animate-ping opacity-20"
   237→                    style={{ borderColor: isDisputed ? '#EF4444' : PARROT_GREEN }}
   238→                  />
   239→                  <span
   240→                    className="absolute -inset-1.5 rounded-full border-2 animate-pulse"
   241→                    style={{ borderColor: isDisputed ? 'rgba(239,68,68,0.25)' : 'rgba(101,163,13,0.25)' }}
   242→                  />
   243→                </>

   245→              <div
   246→                className="relative flex h-10 w-10 items-center justify-center rounded-full border-[3px] transition-all duration-300"
   247→                style={{
   248→                  backgroundColor: isCompleted || isActive
   249→                    ? (isDisputed && isActive ? '#FEE2E2' : PARROT_GREEN)

   252→                    ? (isDisputed && isActive ? '#EF4444' : PARROT_GREEN)

   259→                  <Check className="h-4 w-4" strokeWidth={3} />

   261→                  <AlertTriangle className="h-4 w-4" />

   263→                  <Icon className="h-4 w-4" />
   264→                )}
   265→              </div>
   266→            </div>
   267→            <p
   268→              className="mt-3 text-center text-[11px] font-semibold leading-tight"
   269→              style={{ color: isCompleted || isActive ? 'var(--foreground)' : 'var(--muted-foreground)' }}
   270→            >
   271→              {step.label}
   272→            </p>
   273→          </div>
   274→        );
   275→      })}
   276→    </div>
   277→  );
   278→}
