   338→function PaymentVerifyPanel() {
   339→  const [deals, setDeals] = useState<DealRow[]>([]);
   340→  const [loading, setLoading] = useState(true);
   341→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   342→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   343→  const [confirmDialog, setConfirmDialog] = useState<{
   344→    type: 'verify' | 'reject';
   345→    deal: DealRow;
   346→  } | null>(null);
   347→
   348→  const fetchDeals = useCallback(async () => {
   349→    try {
   350→      const res = await fetch('/api/admin/deals');
   351→      if (res.ok) {
   352→        const data = await res.json();
   353→        setDeals(data);
   354→      }
   355→    } catch {
   356→      // use fallback
   357→    } finally {
   358→      setLoading(false);
   359→    }
   360→  }, []);
   361→
   362→  useEffect(() => {
   363→    fetchDeals();
   364→  }, [fetchDeals]);
   365→
   366→  const handleVerify = async () => {
   367→    if (!confirmDialog) return;
   368→    const dealId = confirmDialog.deal.id;
   369→    setActionLoading(dealId);
   370→    try {
   371→      const res = await fetch('/api/admin/verify', {
   372→        method: 'POST',
   373→        headers: { 'Content-Type': 'application/json' },
   374→        body: JSON.stringify({ dealId }),
   375→      });
   376→      if (res.ok) {
   377→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   378→        setConfirmDialog(null);
   379→        setSelectedDeal(null);
   380→        fetchDeals();
   381→      } else {
   382→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   383→        setConfirmDialog(null);
   384→      }
   385→    } catch {
   386→      toast.error('সার্ভারে সমস্যা');
   387→      setConfirmDialog(null);
   388→    } finally {
   389→      setActionLoading(null);
   390→    }
   391→  };
   392→
   393→  const handleReject = async () => {
   394→    if (!confirmDialog) return;
   395→    const dealId = confirmDialog.deal.id;
   396→    setActionLoading(dealId);
   397→    try {
   398→      const res = await fetch('/api/admin/reject', {
   399→        method: 'POST',
   400→        headers: { 'Content-Type': 'application/json' },
   401→        body: JSON.stringify({ dealId }),
   402→      });
   403→      if (res.ok) {
   404→        toast.success('ডিল রিজেক্ট করা হয়েছে');
   405→        setConfirmDialog(null);
   406→        setSelectedDeal(null);
   407→        fetchDeals();
   408→      } else {
   409→        toast.error('রিজেক্ট ব্যর্থ হয়েছে');
   410→        setConfirmDialog(null);
   411→      }
   412→    } catch {
   413→      toast.error('সার্ভারে সমস্যা');
   414→      setConfirmDialog(null);
   415→    } finally {
   416→      setActionLoading(null);
   417→    }
   418→  };
   419→
   420→  const pendingDeals = deals.filter(
   421→    (d) =>
   422→      d.status === 'created' ||
   423→      d.status === 'pending' ||
   424→      d.status === 'payment_pending'
   425→  );
   426→
   427→  /* ─── Detail View (shown when a deal is selected) ─── */
   428→  if (selectedDeal) {
   429→    const deal = selectedDeal;
   430→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   431→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   432→    const isActing = actionLoading === deal.id;
   433→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   434→      year: 'numeric',
   435→      month: 'short',
   436→      day: 'numeric',
   437→      hour: '2-digit',
   438→      minute: '2-digit',
   439→    });
   440→
   441→    return (
   442→      <div>
   443→        {/* Back button + header */}
   444→        <div className="mb-4 flex items-center gap-2">
   445→          <button
   446→            onClick={() => setSelectedDeal(null)}
   447→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   448→          >
   449→            <ArrowLeft className="h-4 w-4" />
   450→            ফিরে যান
   451→          </button>
   452→        </div>
   453→
   454→        <SolidCard className="!p-0 overflow-hidden">
   455→          {/* Header */}
   456→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   457→            <div className="flex items-center justify-between mb-2">
   458→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   459→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   460→                পেন্ডিং
   461→              </Badge>
   462→            </div>
   463→            <p className="text-base font-bold text-foreground">{userName}</p>
   464→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
   465→          </div>
   466→

   468→          <div className="p-4 sm:p-5 space-y-4">
