   285→export function DealWorkflowTracker() {
   286→  const { setDashboardPanel, activeDeal, user, setActiveDeal } = useAppStore();
   287→  const mounted = useSyncExternalStore(emptySubscribe, () => true, () => false);
   288→
   289→  const [dealData, setDealData] = useState<DealData | null>(null);
   290→  const [loading, setLoading] = useState(true);
   291→  const [actionLoading, setActionLoading] = useState(false);
   292→
   293→  const fetchDeal = useCallback(async () => {
   294→    if (!activeDeal?.id) {
   295→      setLoading(false);
   296→      return;
   297→    }
   298→    setLoading(true);
   299→    try {
   300→      const res = await fetch(`/api/deals/${encodeURIComponent(activeDeal.id)}`);
   301→      if (res.ok) {
   302→        const data: DealData = await res.json();
   303→        setDealData(data);
   304→        if (data.status !== activeDeal.status) {
   305→          setActiveDeal({
   306→            id: data.id,
   307→            title: data.title,
   308→            amount: data.amount,
   309→            status: data.status,
   310→            createdAt: data.createdAt,
   311→            buyerId: data.buyerId,
   312→            sellerId: data.sellerId || undefined,
   313→            creatorId: data.creatorId,
   314→            buyerName: data.buyer?.name,
   315→            sellerName: data.seller?.name,
   316→            rejectionReason: data.rejectionReason || undefined,
   317→          });
   318→        }
   319→      }
   320→    } catch {
   321→      // Use store data as fallback
   322→    } finally {
   323→      setLoading(false);
   324→    }
   325→  }, [activeDeal?.id, activeDeal?.status, setActiveDeal]);
   326→
   327→  useEffect(() => {
   328→    fetchDeal();
   329→  }, [fetchDeal]);
   330→
   331→  if (!mounted) return null;
   332→
   333→  const status = dealData?.status || activeDeal?.status || 'created';
   334→  const activeStep = statusToActiveStep(status);
   335→  const isDisputed = status === 'disputed';
   336→  const isCancelled = status === 'cancelled';
   337→  const isHardRejected = status === 'rejected';
   338→  const isPendingPayment = status === 'Pending_Payment';
   339→  const isCompleted = status === 'completed';
   340→  const canRepay = isPendingPayment || isHardRejected;
   341→
   342→  const dealAmount = dealData?.amount ?? activeDeal?.amount ?? 0;
   343→  const dealTitle = dealData?.title || activeDeal?.title || 'ডিল';
   344→  const dealDate = (dealData?.createdAt || activeDeal?.createdAt)
   345→    ? new Date(dealData?.createdAt || activeDeal?.createdAt || '').toLocaleDateString('bn-BD', { year: 'numeric', month: 'short', day: 'numeric' })
   346→    : '---';
   347→  const buyerName = dealData?.buyer?.name || activeDeal?.buyerName || user?.name || 'ক্রেতা';
   348→  const sellerName = dealData?.seller?.name || activeDeal?.sellerName || 'বিক্রেতা';
   349→  const dealTerms = dealData?.terms;
   350→  const rejectionReason = dealData?.rejectionReason || activeDeal?.rejectionReason || null;
   351→
   352→  /* ── Action Handlers ── */
   353→  const handlePayNow = () => {
   354→    setDashboardPanel('payment');
   355→  };
   356→
   357→  const handleComplete = async () => {
   358→    if (!dealData) return;
   359→    setActionLoading(true);
   360→    try {
   361→      const res = await fetch('/api/deals/complete', {
   362→        method: 'POST',
   363→        headers: { 'Content-Type': 'application/json' },
   364→        body: JSON.stringify({ dealId: dealData.id }),
   365→      });
   366→      if (res.ok) {
   367→        toast.success('ডিলটি সফলভাবে সম্পন্ন হয়েছে!');
   368→        fetchDeal();
   369→      } else {
   370→        const data = await res.json().catch(() => ({}));
   371→        toast.error(data.error || 'আপডেট ব্যর্থ হয়েছে');
   372→      }
   373→    } catch {
   374→      toast.error('নেটওয়ার্ক সমস্যা');
   375→    } finally {
   376→      setActionLoading(false);
   377→    }
   378→  };
   379→
   380→  const handleDispute = async () => {
   381→    if (!dealData) return;
   382→    setActionLoading(true);
   383→    try {
   384→      const res = await fetch('/api/deals/dispute', {
   385→        method: 'POST',
   386→        headers: { 'Content-Type': 'application/json' },
   387→        body: JSON.stringify({ dealId: dealData.id }),
   388→      });
   389→      if (res.ok) {
   390→        toast.success('বিরোধ দায়ের করা হয়েছে। অ্যাডমিন পর্যালোচনা করবেন।');
   391→        fetchDeal();
   392→      } else {
   393→        const data = await res.json().catch(() => ({}));
   394→        toast.error(data.error || 'বিরোধ দায়ের ব্যর্থ হয়েছে');
   395→      }
   396→    } catch {
   397→      toast.error('নেটওয়ার্ক সমস্যা');
   398→    } finally {
   399→      setActionLoading(false);
   400→    }
   401→  };
   402→
   403→  /* ── Loading ── */
   404→  if (loading) {
   405→    return (
   406→      <div className="flex items-center justify-center py-20">
   407→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />

   409→    );
   410→  }
   411→
   412→  return (
   413→    <motion.div
   414→      initial={{ opacity: 0, y: 16 }}
   415→      animate={{ opacity: 1, y: 0 }}
   416→      transition={{ duration: 0.4 }}
   417→      className="mx-auto w-full max-w-2xl"
   418→    >
   419→      <div className="overflow-hidden rounded-xl border shadow-lg" style={{ backgroundColor: 'var(--background, #fff)', borderColor: 'var(--border)' }}>
   420→        {/* ── Header ── */}
   421→        <div className="flex items-center justify-between border-b px-4 py-3 sm:px-5" style={{ borderColor: 'var(--border)' }}>
   422→          <div className="flex items-center gap-3">
   423→            <button
   424→              onClick={() => setDashboardPanel('my-deals')}
   425→              className="flex h-8 w-8 items-center justify-center rounded-lg transition-colors hover:bg-accent"
   426→              aria-label="ফিরে যান"
   427→            >
   428→              <ArrowLeft className="h-4 w-4" style={{ color: 'var(--foreground)' }} />
   429→            </button>
   430→            <div className="flex items-center gap-2">
   431→              <div className="flex h-8 w-8 items-center justify-center rounded-lg shadow-md" style={{ backgroundColor: '#84CC16' }}>
   432→                <span className="text-sm font-bold" style={{ color: '#18181b' }}>আ</span>
   433→              </div>
   434→              <span className="text-base font-bold tracking-tight" style={{ color: 'var(--foreground)' }}>
   435→                ডিল বিবরণ
   436→              </span>
   437→            </div>
   438→          </div>
   439→
   440→          <div className="flex items-center gap-2">
   441→            <span className="hidden sm:block text-xs font-medium truncate max-w-[100px]" style={{ color: 'var(--muted-foreground)' }}>
   442→              {user?.name || 'ব্যবহারকারী'}
   443→            </span>
