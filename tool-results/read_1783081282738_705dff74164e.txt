    57→    case 'Payment_Verified': return 2;

   296→export function DealWorkflowTracker() {
   297→  const { setDashboardPanel, activeDeal, user, setActiveDeal } = useAppStore();
   298→  const mounted = useSyncExternalStore(emptySubscribe, () => true, () => false);
   299→
   300→  const [dealData, setDealData] = useState<DealData | null>(null);
   301→  const [loading, setLoading] = useState(true);
   302→  const [actionLoading, setActionLoading] = useState(false);
   303→  const [messages, setMessages] = useState<ChatMsg[]>([]);
   304→  const messagesEndRef = useRef<HTMLDivElement>(null);
   305→
   306→  const fetchDeal = useCallback(async () => {
   307→    if (!activeDeal?.id) {
   308→      setLoading(false);
   309→      return;
   310→    }
   311→    setLoading(true);
   312→    try {
   313→      const res = await fetch(`/api/deals/${encodeURIComponent(activeDeal.id)}`);
   314→      if (res.ok) {
   315→        const data: DealData = await res.json();
   316→        setDealData(data);
   317→        if (data.status !== activeDeal.status) {
   318→          setActiveDeal({
   319→            id: data.id,
   320→            title: data.title,
   321→            amount: data.amount,
   322→            status: data.status,
   323→            createdAt: data.createdAt,
   324→            buyerId: data.buyerId,
   325→            sellerId: data.sellerId || undefined,
   326→            creatorId: data.creatorId,
   327→            buyerName: data.buyer?.name,
   328→            sellerName: data.seller?.name,
   329→            rejectionReason: data.rejectionReason || undefined,
   330→          });
   331→        }
   332→      }
   333→    } catch {
   334→      // Use store data as fallback
   335→    } finally {
   336→      setLoading(false);
   337→    }
   338→  }, [activeDeal?.id, activeDeal?.status, setActiveDeal]);
   339→
   340→  const fetchMessages = useCallback(async () => {
   341→    if (!activeDeal?.id) return;
   342→    try {
   343→      const res = await fetch(`/api/deals/${encodeURIComponent(activeDeal.id)}/chat`);
   344→      if (res.ok) {
   345→        const data: ChatMsg[] = await res.json();
   346→        setMessages(data);
   347→      }
   348→    } catch { /* silent */ }
   349→  }, [activeDeal?.id]);
   350→
   351→  useEffect(() => {
   352→    fetchDeal();
   353→    fetchMessages();
   354→  }, [fetchDeal, fetchMessages]);
   355→
   356→  // Auto-scroll messages to bottom
   357→  useEffect(() => {
   358→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   359→  }, [messages]);
   360→
   361→  if (!mounted) return null;
   362→
   363→  const status = dealData?.status || activeDeal?.status || 'created';
   364→  const activeStep = statusToActiveStep(status);
   365→  const isDisputed = status === 'disputed';
   366→  const isCancelled = status === 'cancelled';
   367→  const isHardRejected = status === 'rejected';
   368→  const isPendingPayment = status === 'Pending_Payment';
   369→  const isCompleted = status === 'completed';
   370→  const canRepay = isPendingPayment || isHardRejected;
   371→
   372→  const dealAmount = dealData?.amount ?? activeDeal?.amount ?? 0;
   373→  const dealTitle = dealData?.title || activeDeal?.title || 'ডিল';
   374→  const dealDate = (dealData?.createdAt || activeDeal?.createdAt)
   375→    ? new Date(dealData?.createdAt || activeDeal?.createdAt || '').toLocaleDateString('bn-BD', { year: 'numeric', month: 'short', day: 'numeric' })
   376→    : '---';
   377→  const buyerName = dealData?.buyer?.name || activeDeal?.buyerName || user?.name || 'ক্রেতা';
   378→  const sellerName = dealData?.seller?.name || activeDeal?.sellerName || 'বিক্রেতা';
   379→  const dealTerms = dealData?.terms;
   380→  const rejectionReason = dealData?.rejectionReason || activeDeal?.rejectionReason || null;
   381→
   382→  /* ── Action Handlers ── */
   383→  const handlePayNow = () => {
   384→    setDashboardPanel('payment');
   385→  };
   386→
   387→  const handleComplete = async () => {
   388→    if (!dealData) return;
   389→    setActionLoading(true);
   390→    try {
   391→      const res = await fetch('/api/deals/complete', {
   392→        method: 'POST',
   393→        headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({ dealId: dealData.id }),
   395→      });
   396→      if (res.ok) {
   397→        toast.success('ডিলটি সফলভাবে সম্পন্ন হয়েছে!');
   398→        fetchDeal();
   399→        fetchMessages();
   400→      } else {
   401→        const data = await res.json().catch(() => ({}));
   402→        toast.error(data.error || 'আপডেট ব্যর্থ হয়েছে');
   403→      }
   404→    } catch {
   405→      toast.error('নেটওয়ার্ক সমস্যা');
   406→    } finally {
   407→      setActionLoading(false);
   408→    }
   409→  };
   410→
   411→  const handleDispute = async () => {
   412→    if (!dealData) return;
   413→    setActionLoading(true);
   414→    try {
   415→      const res = await fetch('/api/deals/dispute', {
   416→        method: 'POST',
   417→        headers: { 'Content-Type': 'application/json' },
   418→        body: JSON.stringify({ dealId: dealData.id }),
   419→      });
   420→      if (res.ok) {
   421→        toast.success('বিরোধ দায়ের করা হয়েছে। অ্যাডমিন পর্যালোচনা করবেন।');
   422→        fetchDeal();
   423→        fetchMessages();
   424→      } else {
   425→        const data = await res.json().catch(() => ({}));
   426→        toast.error(data.error || 'বিরোধ দায়ের ব্যর্থ হয়েছে');
   427→      }
   428→    } catch {
   429→      toast.error('নেটওয়ার্ক সমস্যা');
   430→    } finally {
   431→      setActionLoading(false);
   432→    }
   433→  };
   434→
   435→  /* ── Loading ── */
   436→  if (loading) {
   437→    return (
   438→      <div className="flex items-center justify-center py-20">
   439→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   440→      </div>
   441→    );
   442→  }
   443→
   444→  return (
   445→    <motion.div
   446→      initial={{ opacity: 0, y: 16 }}
   447→      animate={{ opacity: 1, y: 0 }}
   448→      transition={{ duration: 0.4 }}
   449→      className="mx-auto w-full max-w-2xl"
   450→    >
   451→      <div className="overflow-hidden rounded-xl border shadow-lg" style={{ backgroundColor: 'var(--background, #fff)', borderColor: 'var(--border)' }}>
   452→        {/* ── Header ── */}
   453→        <div className="flex items-center justify-between border-b px-4 py-3 sm:px-5" style={{ borderColor: 'var(--border)' }}>
   454→          <div className="flex items-center gap-3">
   455→            <button
   456→              onClick={() => setDashboardPanel('my-deals')}
   457→              className="flex h-8 w-8 items-center justify-center rounded-lg transition-colors hover:bg-accent"
