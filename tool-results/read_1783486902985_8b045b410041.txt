   324→function PaymentVerifyPanel() {
   325→  const [deals, setDeals] = useState<DealRow[]>([]);
   326→  const [loading, setLoading] = useState(true);
   327→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   328→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   329→  const [confirmDialog, setConfirmDialog] = useState<{
   330→    type: 'verify' | 'reject';
   331→    deal: DealRow;
   332→  } | null>(null);
   333→
   334→  const fetchDeals = useCallback(async () => {
   335→    try {
   336→      const res = await fetch('/api/admin/deals');
   337→      if (res.ok) {
   338→        const data = await res.json();
   339→        setDeals(data);
   340→      }
   341→    } catch {
   342→      // use fallback
   343→    } finally {
   344→      setLoading(false);
   345→    }
   346→  }, []);
   347→
   348→  useEffect(() => {
   349→    fetchDeals();
   350→  }, [fetchDeals]);
   351→
   352→  const handleVerify = async () => {
   353→    if (!confirmDialog) return;
   354→    const dealId = confirmDialog.deal.id;
   355→    setActionLoading(dealId);
   356→    try {
   357→      const res = await fetch('/api/admin/verify', {
   358→        method: 'POST',
   359→        headers: { 'Content-Type': 'application/json' },
   360→        body: JSON.stringify({ dealId }),
   361→      });
   362→      if (res.ok) {
   363→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   364→        setConfirmDialog(null);
   365→        setSelectedDeal(null);
   366→        fetchDeals();
   367→      } else {
   368→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   369→        setConfirmDialog(null);
   370→      }
   371→    } catch {
   372→      toast.error('সার্ভারে সমস্যা');
   373→      setConfirmDialog(null);
   374→    } finally {
   375→      setActionLoading(null);
   376→    }
   377→  };
   378→
   379→  const handleReject = async () => {
   380→    if (!confirmDialog) return;
   381→    const dealId = confirmDialog.deal.id;
   382→    setActionLoading(dealId);
   383→    try {
   384→      const res = await fetch('/api/admin/reject', {
   385→        method: 'POST',
   386→        headers: { 'Content-Type': 'application/json' },
   387→        body: JSON.stringify({ dealId }),
   388→      });
   389→      if (res.ok) {
   390→        toast.success('ডিল রিজেক্ট করা হয়েছে');
   391→        setConfirmDialog(null);
   392→        setSelectedDeal(null);
   393→        fetchDeals();
   394→      } else {
   395→        toast.error('রিজেক্ট ব্যর্থ হয়েছে');
   396→        setConfirmDialog(null);
   397→      }
   398→    } catch {
   399→      toast.error('সার্ভারে সমস্যা');
   400→      setConfirmDialog(null);
   401→    } finally {
   402→      setActionLoading(null);
   403→    }
   404→  };
   405→
   406→  const pendingDeals = deals.filter(
   407→    (d) =>
   408→      d.status === 'created' ||
   409→      d.status === 'pending' ||
   410→      d.status === 'payment_pending'
   411→  );
   412→
   413→  /* ─── Detail View (shown when a deal is selected) ─── */
   414→  if (selectedDeal) {
   415→    const deal = selectedDeal;
   416→    const shortId = deal.id.length > 10 ? 'DL-' + deal.id.slice(-5) : deal.id;
   417→    const userName = deal.creator?.name || deal.buyer?.name || 'ইউজার';
   418→    const isActing = actionLoading === deal.id;
   419→    const dealTime = new Date(deal.createdAt).toLocaleString('bn-BD', {
   420→      year: 'numeric',
   421→      month: 'short',
   422→      day: 'numeric',
   423→      hour: '2-digit',
   424→      minute: '2-digit',
   425→    });
   426→
   427→    return (
   428→      <div>
   429→        {/* Back button + header */}
   430→        <div className="mb-4 flex items-center gap-2">
   431→          <button
   432→            onClick={() => setSelectedDeal(null)}
   433→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   434→          >
   435→            <ArrowLeft className="h-4 w-4" />
   436→            ফিরে যান
   437→          </button>
   438→        </div>
   439→
   440→        <SolidCard className="!p-0 overflow-hidden">
   441→          {/* Header */}
   442→          <div className="border-b border-border/50 bg-muted/30 px-4 sm:px-5 py-4">
   443→            <div className="flex items-center justify-between mb-2">
   444→              <span className="font-mono text-xs text-muted-foreground">{shortId}</span>
   445→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400 border-0 text-xs font-medium">
   446→                পেন্ডিং
   447→              </Badge>
   448→            </div>
   449→            <p className="text-base font-bold text-foreground">{userName}</p>
   450→            <p className="text-2xl font-extrabold text-foreground mt-1">৳{deal.amount.toLocaleString('bn-BD')}</p>
   451→          </div>
   452→
   453→          {/* Payment Proof Details */}
   454→          <div className="p-4 sm:p-5 space-y-4">
   455→            {/* Payment Method */}
   456→            {deal.paymentMethod && (
