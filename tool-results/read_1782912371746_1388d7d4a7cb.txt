   177→/* ─── Admin Notification Bell ─── */
   178→function AdminNotificationBell() {
   179→  const user = useAppStore((s) => s.user);
   180→  const [open, setOpen] = useState(false);
   181→  const [notifications, setNotifications] = useState<any[]>([]);
   182→  const [unreadCount, setUnreadCount] = useState(0);
   183→  const [loading, setLoading] = useState(true);
   184→  const dropdownRef = useRef<HTMLDivElement>(null);
   185→
   186→  const { onNotification } = useSocket('admin');
   187→
   188→  const fetchNotifs = useCallback(async () => {
   189→    try {
   190→      const res = await fetch('/api/notifications?userId=admin');
   191→      if (res.ok) {
   192→        const data = await res.json();
   193→        setNotifications(data.notifications || []);
   194→        setUnreadCount(data.unreadCount || 0);
   195→      }
   196→    } catch { /* silent */ }
   197→    finally { setLoading(false); }
   198→  }, []);
   199→
   200→  useEffect(() => { fetchNotifs() }, [fetchNotifs]);
   201→
   202→  useEffect(() => {
   203→    const unsub = onNotification(() => {
   204→      fetchNotifs();
   205→      toast.success('নতুন নোটিফিকেশন!');
   206→    });
   207→    return unsub;
   208→  }, [onNotification, fetchNotifs]);
   209→
   210→  useEffect(() => {
   211→    const handleClick = (e: MouseEvent) => {
   212→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   213→        setOpen(false);
   214→      }
   215→    };
   216→    document.addEventListener('mousedown', handleClick);
   217→    return () => document.removeEventListener('mousedown', handleClick);
   218→  }, []);
   219→
   220→  return (
   221→    <div className="relative" ref={dropdownRef}>
   222→      <button
   223→        onClick={() => setOpen(!open)}
   224→        className="relative flex h-10 w-10 items-center justify-center rounded-xl text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   225→        aria-label="নোটিফিকেশন"
   226→      >
   227→        <Bell className="h-5 w-5" />
   228→        {unreadCount > 0 && (
   229→          <span className="absolute -top-0.5 -right-0.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-red-500 px-1 text-[10px] font-bold text-white shadow-sm">
   230→            {unreadCount > 99 ? '৯৯+' : unreadCount}
   231→          </span>
   232→        )}
   233→      </button>
   234→
   235→      <AnimatePresence>
   236→        {open && (
   237→          <motion.div
   238→            initial={{ opacity: 0, y: -8, scale: 0.95 }}
   239→            animate={{ opacity: 1, y: 0, scale: 1 }}
   240→            exit={{ opacity: 0, y: -8, scale: 0.95 }}
   241→            transition={{ duration: 0.2 }}
   242→            className="absolute right-0 top-12 z-50 w-80 rounded-2xl bg-white shadow-2xl border border-border/50 dark:bg-zinc-900 overflow-hidden"
   243→          >
   244→            <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   245→              <h3 className="text-sm font-bold text-foreground">অ্যাডমিন নোটিফিকেশন</h3>
   246→            </div>
   247→            <div className="max-h-72 overflow-y-auto">
   248→              {loading ? (
   249→                <div className="flex items-center justify-center py-10">
   250→                  <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   251→                </div>
   252→              ) : notifications.length === 0 ? (
   253→                <div className="flex flex-col items-center py-10 text-center">
   254→                  <Bell className="h-8 w-8 text-muted-foreground/40 mb-2" />
   255→                  <p className="text-sm text-muted-foreground">কোনো নোটিফিকেশন নেই</p>
   256→                </div>
   257→              ) : (
   258→                notifications.map((notif: any) => (
   259→                  <div
   260→                    key={notif.id}
   261→                    className={`px-4 py-3 border-b border-border/30 last:border-0 ${
   262→                      !notif.read ? 'bg-primary/5' : ''
   263→                    }`}
   264→                  >
   265→                    <div className="flex items-start gap-2">
   266→                      {!notif.read && <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />}
   267→                      <div className="min-w-0 flex-1">
   268→                        <p className="text-xs font-semibold text-foreground">{notif.title}</p>
   269→                        <p className="mt-0.5 text-[11px] text-muted-foreground line-clamp-2">{notif.message}</p>
   270→                      </div>
   271→                    </div>
   272→                  </div>
   273→                ))
   274→              )}
   275→            </div>
   276→          </motion.div>
   277→        )}
   278→      </AnimatePresence>
   279→    </div>
   280→  );
   281→}

   393→      // use fallback

   395→      setLoading(false);
