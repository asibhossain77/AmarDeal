    73→

    81→  const [deletingId, setDeletingId] = useState<string | null>(null);
    82→

   662→function getContrastColor(hex: string): string {
   663→  const r = parseInt(hex.slice(1, 3), 16);
   664→  const g = parseInt(hex.slice(3, 5), 16);
   665→  const b = parseInt(hex.slice(5, 7), 16);
   666→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   667→  return luminance > 0.6 ? '#18181b' : '#ffffff';
   668→}
