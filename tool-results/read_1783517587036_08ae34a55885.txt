    65→/* ─── Admin Chat Message Styles ─── */
    66→function adminChatBubble(msg: AdminChatMsg) {
    67→  const isAdmin = msg.role === 'admin';
    68→  const isSystem = msg.role === 'system';
    69→  const isBuyer = msg.role === 'buyer';
    70→  const isSeller = msg.role === 'seller';
    71→
    72→  const align = isAdmin || isSystem ? 'justify-center' : isBuyer ? 'justify-start' : 'justify-end';
    73→
    74→  const bubbleStyle = isSystem
    75→    ? 'bg-amber-50 dark:bg-amber-500/8 text-amber-800 dark:text-amber-200 border border-amber-200/60 dark:border-amber-500/20'
    76→    : isAdmin
    77→    ? 'bg-purple-50 dark:bg-purple-500/8 text-purple-900 dark:text-purple-200 border border-purple-200/60 dark:border-purple-500/20'
    78→    : isBuyer
    79→    ? 'bg-sky-50 dark:bg-sky-500/8 text-sky-900 dark:text-sky-200 border border-sky-200/60 dark:border-sky-500/20'
    80→    : 'bg-emerald-50 dark:bg-emerald-500/8 text-emerald-900 dark:text-emerald-200 border border-emerald-200/60 dark:border-emerald-500/20';
    81→
    82→  const labelStyle = isSystem
    83→    ? 'text-[10px] font-bold text-amber-600 dark:text-amber-400'
    84→    : isAdmin
    85→    ? 'text-[10px] font-bold text-purple-600 dark:text-purple-400'
    86→    : isBuyer
    87→    ? 'text-[10px] font-bold text-sky-600 dark:text-sky-400'
    88→    : 'text-[10px] font-bold text-emerald-600 dark:text-emerald-400';
    89→
    90→  const label = isSystem ? 'সিস্টেম' : isAdmin ? `${msg.senderName} (অ্যাডমিন)` : msg.senderName || 'অজানা';
    91→  const showLabel = isSystem || isAdmin || isBuyer || isSeller;
    92→
    93→  return (
    94→    <div key={msg.id} className={`flex ${align}`}>
    95→      <div className={`max-w-[80%] rounded-2xl px-4 py-2.5 text-sm ${bubbleStyle}`}>
    96→        {showLabel && <p className={`${labelStyle} mb-0.5`}>{label}</p>}
    97→        <p className="leading-relaxed">{msg.text}</p>
    98→      </div>
    99→    </div>
   100→  );
   101→}

   386→function PaymentVerifyPanel() {
   387→  const [deals, setDeals] = useState<DealRow[]>([]);
   388→  const [loading, setLoading] = useState(true);
   389→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   390→  const [selectedDeal, setSelectedDeal] = useState<DealRow | null>(null);
   391→  const [confirmDialog, setConfirmDialog] = useState<{
   392→    type: 'verify' | 'reject';
   393→    deal: DealRow;
   394→  } | null>(null);
   395→
   396→  const fetchDeals = useCallback(async () => {
   397→    try {
   398→      const res = await fetch('/api/admin/deals');
   399→      if (res.ok) {
   400→        const data = await res.json();
   401→        setDeals(data);
   402→      }
   403→    } catch {
   404→      // use fallback
   405→    } finally {
   406→      setLoading(false);
   407→    }
   408→  }, []);
   409→
   410→  useEffect(() => {
   411→    fetchDeals();
   412→  }, [fetchDeals]);
   413→
   414→  const handleVerify = async () => {
   415→    if (!confirmDialog) return;
   416→    const dealId = confirmDialog.deal.id;
   417→    setActionLoading(dealId);
   418→    try {
   419→      const res = await fetch('/api/admin/verify', {
   420→        method: 'POST',
   421→        headers: { 'Content-Type': 'application/json' },
   422→        body: JSON.stringify({ dealId }),
   423→      });
   424→      if (res.ok) {
   425→        toast.success('পেমেন্ট সফলভাবে ভেরিফাই হয়েছে!');
   426→        setConfirmDialog(null);
   427→        setSelectedDeal(null);
   428→        fetchDeals();
   429→      } else {
   430→        toast.error('ভেরিফিকেশন ব্যর্থ হয়েছে');
   431→        setConfirmDialog(null);
   432→      }
   433→    } catch {
   434→      toast.error('সার্ভারে সমস্যা');
   435→      setConfirmDialog(null);
   436→    } finally {
   437→      setActionLoading(null);
   438→    }
   439→  };
   440→
   441→  const handleReject = async () => {
   442→    if (!confirmDialog) return;
   443→    const dealId = confirmDialog.deal.id;
   444→    setActionLoading(dealId);
   445→    try {
   446→      const res = await fetch('/api/admin/reject', {
   447→        method: 'POST',
   448→        headers: { 'Content-Type': 'application/json' },
   449→        body: JSON.stringify({ dealId }),
   450→      });
   451→      if (res.ok) {
   452→        toast.success('ডিল রিজেক্ট করা হয়েছে');
   453→        setConfirmDialog(null);
   454→        setSelectedDeal(null);

   459→      }
   460→    } catch {
   461→      toast.error('সার্ভারে সমস্যা');
   462→      setConfirmDialog(null);

   464→      setActionLoading(null);

   486→      minute: '2-digit',
   487→    });
   488→
   489→    return (
   490→      <div>
   491→        {/* Back button + header */}
   492→        <div className="mb-4 flex items-center gap-2">
   493→          <button
   494→            onClick={() => setSelectedDeal(null)}
   495→            className="flex items-center gap-1 text-sm font-medium text-muted-foreground hover:text-foreground transition-colors"
   496→          >
   497→            <ArrowLeft className="h-4 w-4" />
   498→            ফিরে যান
